import sys
import bisect

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    W = int(next(it))

    blocks = []
    for _ in range(N):
        x = int(next(it))
        y = int(next(it))
        blocks.append((x, y))

    Q = int(next(it))
    queries = []
    for _ in range(Q):
        t = int(next(it))
        a = int(next(it)) - 1
        queries.append((t, a))

    # Group blocks by column
    col_blocks = [[] for _ in range(W + 1)]
    for idx, (x, y) in enumerate(blocks):
        col_blocks[x].append((y, idx))

    # Sort each column by y descending (topmost first)
    for lst in col_blocks:
        lst.sort(reverse=True)

    # For each column, compute for each block the time it disappears
    disappear_time = [0] * N
    for x in range(1, W + 1):
        col = col_blocks[x]
        m = len(col)
        # Process from bottom to top
        for i in range(m):
            y, idx = col[i]
            # The bottom block in this column disappears at time y
            if i == m - 1:
                disappear_time[idx] = y
            else:
                # The block above disappears when the block below is gone
                # and it has fallen to the bottom row.
                # Actually, the disappearance time is max(y, disappear_time of below + 1)
                # because it must wait for the block below to disappear before it can fall.
                below_y, below_idx = col[i + 1]
                # The block below disappears at disappear_time[below_idx]
                # This block will start falling after that, and it needs (y - below_y - 1) steps?
                # Let's think differently:
                # The bottom block disappears at time y (its own y).
                # The block just above disappears at time y + 1? Not exactly.
                # We can simulate by "stack" of disappearance times.
                # Actually, the k-th block from bottom disappears at y + (k-1).
                # Because after the bottom disappears, the next falls one step per time unit.
                # But careful: the bottom block disappears at time y, then the next block
                # becomes bottom at time y+1 and disappears at time (y+1) + (new y?) Wait,
                # its y coordinate at time y+1 is y (since it fell one step).
                # So it disappears at time y+1.
                # So indeed, if blocks are sorted by y increasing (bottom to top),
                # disappearance times are y, y+1, y+2, ...
                # But here we have descending order. Let's convert.
                pass

    # Better approach: sort each column by y ascending (bottom first)
    for x in range(1, W + 1):
        col = col_blocks[x]
        col.sort()  # now y ascending
        m = len(col)
        for k in range(m):
            y, idx = col[k]
            disappear_time[idx] = y + k

    # Now answer queries
    out_lines = []
    for t, a in queries:
        if t < disappear_time[a]:
            out_lines.append("Yes")
        else:
            out_lines.append("No")
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
