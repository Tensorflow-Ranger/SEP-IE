def iscube(a):
    '''
    Write a function that takes an integer a and returns True 
    if this integer has an integer root.
    '''
    if a < 0:
        a = -a
    if a == 0:
        return True
    for i in range(1, int(a ** (1/3)) + 2):
        if i ** 3 == a:
            return True
    return False
