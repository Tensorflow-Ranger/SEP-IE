from typing import List, Tuple


def find_closest_elements(numbers: List[float]) -> Tuple[float, float]:
    """ From a supplied list of numbers (of length at least two) select and return two that are the nearest neighbors and return them in order (smaller number, larger number).
    """
    if len(numbers) < 2:
        raise ValueError("List must contain at least two numbers")
    
    sorted_numbers = sorted(numbers)
    min_diff = float('inf')
    result = (0.0, 0.0)
    
    for i in range(len(sorted_numbers) - 1):
        diff = abs(sorted_numbers[i] - sorted_numbers[i + 1])
        if diff < min_diff:
            min_diff = diff
            result = (sorted_numbers[i], sorted_numbers[i + 1])
    
    return result
