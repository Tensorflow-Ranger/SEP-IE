import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    # Step 1: compress values to 0..N-1
    val_to_idx = {}
    compressed = []
    cur = 0
    for x in A:
        if x not in val_to_idx:
            val_to_idx[x] = cur
            cur += 1
        compressed.append(val_to_idx[x])

    # Step 2: prefix distinct counts
    prefix_distinct = [0] * N
    seen = [False] * cur
    distinct = 0
    for i in range(N):
        idx = compressed[i]
        if not seen[idx]:
            seen[idx] = True
            distinct += 1
        prefix_distinct[i] = distinct

    # Step 3: suffix distinct counts
    suffix_distinct = [0] * N
    seen = [False] * cur
    distinct = 0
    for i in range(N - 1, -1, -1):
        idx = compressed[i]
        if not seen[idx]:
            seen[idx] = True
            distinct += 1
        suffix_distinct[i] = distinct

    # Step 4: middle distinct counts for each possible split
    # We need for each j (end of middle) the distinct count in middle
    # Let's precompute for each i (start of middle) the distinct count up to end j
    # But we can compute on the fly with a sliding window.
    # We'll fix i (start of middle) and move j from i to N-2.
    # Keep a frequency array for current window.
    freq = [0] * cur
    window_distinct = 0
    ans = 0

    # i is the index where middle starts (i from 1 to N-2)
    # j is the index where middle ends (j from i to N-2)
    # We'll slide i from 1 to N-2, and for each i, we expand j gradually.
    j = 1  # j will be the end index of middle (inclusive)
    for i in range(1, N - 1):
        # Ensure window is [i, j] initially when i moves forward
        if j < i:
            j = i
            window_distinct = 0
            freq = [0] * cur
            # add A[i] to window
            idx = compressed[i]
            if freq[idx] == 0:
                window_distinct += 1
            freq[idx] += 1
        else:
            # When i moves forward, remove A[i-1] from window
            idx_remove = compressed[i - 1]
            freq[idx_remove] -= 1
            if freq[idx_remove] == 0:
                window_distinct -= 1

        # Now expand j as far as possible while keeping middle non-empty
        # and j < N-1 (because after j we need at least one element for third part)
        while j < N - 1:
            # try to add A[j+1] to window
            idx_add = compressed[j + 1]
            if freq[idx_add] == 0:
                # adding would increase distinct count
                # but we must ensure middle is non-empty (it already is)
                # and third part is non-empty: j+1 < N-1? Actually third part starts at j+1,
                # so we need j+1 <= N-1, but third part must have at least one element,
                # so j+1 <= N-1 is always true if j < N-1.
                # We can add.
                window_distinct += 1
                freq[idx_add] += 1
                j += 1
            else:
                # adding wouldn't increase distinct count, still we can add
                freq[idx_add] += 1
                j += 1
            # After adding, compute total distinct sum
            total = prefix_distinct[i - 1] + window_distinct + suffix_distinct[j + 1]
            ans = max(ans, total)

        # Also consider current window [i, j] (j might be N-2)
        total = prefix_distinct[i - 1] + window_distinct + suffix_distinct[j + 1]
        ans = max(ans, total)

    print(ans)

if __name__ == "__main__":
    solve()
