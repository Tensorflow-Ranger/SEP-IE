```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    p = list(map(int, data[1:1 + n]))

    pos = [0] * (n + 1)
    for idx, val in enumerate(p):
        pos[val] = idx + 1  # 1‑based index

    total_cost = 0
    for val in range(1, n):
        if pos[val] > pos[val + 1]:
            total_cost += val

    print(total_cost)

if __name__ == "__main__":
    solve()
```
