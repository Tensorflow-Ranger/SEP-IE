def fizz_buzz(n: int):
    """Return the number of times a digit appears in integers less than n which are divisible by 11 or 13.
    """
    from collections import Counter
    
    digit_counts = Counter()
    
    for i in range(n):
        if i % 11 == 0 or i % 13 == 0:
            digit_counts.update(str(i))
    
    return digit_counts
