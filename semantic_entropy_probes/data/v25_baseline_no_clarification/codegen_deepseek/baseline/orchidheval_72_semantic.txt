def will_it_fly(q,w):
    '''
    Write a function that returns True if the object q will fly, and False otherwise.
    The object q will fly when balanced with its weight not exceeding w.
    '''
    # Check if the list is balanced (palindrome)
    if q != q[::-1]:
        return False
    # Check if sum of elements (weight) does not exceed w
    if sum(q) > w:
        return False
    return True
