```python
from typing import List

def parse_music(music_string: str) -> List[int]:
    beat_map = {
        'o': 4,
        'o|': 2,
        '.|': 1
    }
    notes = music_string.split()
    beats = []
    for note in notes:
        if note in beat_map:
            beats.append(beat_map[note])
    return beats
```
