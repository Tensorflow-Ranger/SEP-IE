```python
import sys

def main():
    data = sys.stdin.read().strip().splitlines()
    if not data:
        return
    n = int(data[0])
    strings = data[1:1 + n]
    # Sort by length
    strings.sort(key=len)
    # Concatenate
    result = ''.join(strings)
    print(result)

if __name__ == "__main__":
    main()
```
