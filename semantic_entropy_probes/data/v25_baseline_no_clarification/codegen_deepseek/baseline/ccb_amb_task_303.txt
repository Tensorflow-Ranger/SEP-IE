import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    t = int(next(it))
    out_lines = []
    for _ in range(t):
        n = int(next(it))
        arr = [int(next(it)) for _ in range(2 * n)]
        pos = [[] for _ in range(n + 1)]
        for idx, val in enumerate(arr):
            pos[val].append(idx)
        
        # adjacency list for couples whose occurrences are adjacent
        adj = [[] for _ in range(n + 1)]
        for i in range(1, n + 1):
            p1, p2 = pos[i]
            if abs(p1 - p2) == 1:
                mn, mx = min(p1, p2), max(p1, p2)
                # connect to the couple that starts at mx+1 if it exists?
                # We'll build graph differently: connect i with j if intervals interleave.
                pass
        
        # Build permutation of couples based on first occurrence
        order = list(range(1, n + 1))
        order.sort(key=lambda x: pos[x][0])
        first_pos = {i: pos[i][0] for i in range(1, n + 1)}
        second_pos = {i: pos[i][1] for i in range(1, n + 1)}
        
        # Stack to find nesting
        stack = []
        interleave_edges = []
        for i in order:
            p1 = first_pos[i]
            p2 = second_pos[i]
            while stack and stack[-1][1] < p1:
                stack.pop()
            if stack and stack[-1][1] < p2:
                # interleaving
                j = stack[-1][0]
                interleave_edges.append((j, i))
            stack.append((i, p2))
        
        # Build graph from interleave_edges
        g = [[] for _ in range(n + 1)]
        for u, v in interleave_edges:
            g[u].append(v)
            g[v].append(u)
        
        # Find connected components
        visited = [False] * (n + 1)
        comp_sizes = []
        for i in range(1, n + 1):
            if not visited[i]:
                stack = [i]
                visited[i] = True
                size = 0
                while stack:
                    node = stack.pop()
                    size += 1
                    for nb in g[node]:
                        if not visited[nb]:
                            visited[nb] = True
                            stack.append(nb)
                comp_sizes.append(size)
        
        total_pairs = 0
        for sz in comp_sizes:
            total_pairs += sz * (sz - 1) // 2
        
        out_lines.append(str(total_pairs))
    
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
