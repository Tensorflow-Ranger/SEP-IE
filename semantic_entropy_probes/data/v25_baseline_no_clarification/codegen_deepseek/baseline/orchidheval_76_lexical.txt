def is_simple_power(x, n):
    """Your task is to write a function that returns true if a number x is a natural
    power of n and false in other cases.
    x is a natural power of n if n**int=x
    """
    if x <= 0 or n <= 0:
        return False
    if n == 1:
        return x == 1
    power = 0
    while n ** power < x:
        power += 1
    return n ** power == x
