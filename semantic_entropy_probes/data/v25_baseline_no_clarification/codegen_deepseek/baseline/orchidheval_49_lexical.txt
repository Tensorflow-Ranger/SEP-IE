import sys

def modp(n: int, p: int):
    """Return 2^n over p (be aware of numerics).
    """
    if p == 1:
        return 0
    result = 1
    base = 2 % p
    exp = n
    while exp > 0:
        if exp % 2 == 1:
            result = (result * base) % p
        base = (base * base) % p
        exp //= 2
    return result
