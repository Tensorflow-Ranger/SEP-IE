```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    # Count distinct numbers from left and right
    left_seen = set()
    right_seen = set()
    left_count = [0] * N
    right_count = [0] * N

    for i in range(N):
        left_seen.add(A[i])
        left_count[i] = len(left_seen)

    for i in range(N - 1, -1, -1):
        right_seen.add(A[i])
        right_count[i] = len(right_seen)

    # For each split i (0-based index i means split after element i)
    best = 0
    for i in range(N - 1):
        total = left_count[i] + right_count[i + 1]
        if total > best:
            best = total

    print(best)

if __name__ == "__main__":
    solve()
```
