def words_string(s):
    """
    You will be given a string of words separated by standard delimiters. Your task is
    to split the string into words and return an array of the words.
    """
    import re
    # Split on commas, spaces, or any combination/sequence of them
    if not s:
        return []
    return [word for word in re.split(r'[,\s]+', s) if word]
