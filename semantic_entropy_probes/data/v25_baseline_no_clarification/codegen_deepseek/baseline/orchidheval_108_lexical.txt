def count_nums(arr):
    """
    Write a function count_nums which takes an array of integers and returns
    the number of values which has a sum of digits > 0.
    If a number is negative, then its first signed digit will be negative:
    e.g. -123 has signed digits -1, 2, and 3.
    """
    def digit_sum(n):
        if n == 0:
            return 0
        sign = -1 if n < 0 else 1
        n = abs(n)
        total = 0
        # Convert to string to handle digits easily, but adjust first digit for negatives
        s = str(n)
        for i, ch in enumerate(s):
            digit = int(ch)
            if i == 0 and sign == -1:
                total -= digit
            else:
                total += digit
        return total

    count = 0
    for num in arr:
        if digit_sum(num) > 0:
            count += 1
    return count
