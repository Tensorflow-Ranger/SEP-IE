```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    A = [int(next(it)) for _ in range(N)]
    B = [int(next(it)) for _ in range(N)]
    C = [int(next(it)) for _ in range(N)]

    total = 0
    for i in range(N):
        if A[i] != B[i]:
            total += C[i]
    print(total)

if __name__ == "__main__":
    solve()
```
