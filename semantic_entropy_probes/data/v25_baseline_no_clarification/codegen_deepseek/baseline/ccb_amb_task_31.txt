import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    N = int(data[0])
    A = data[1].strip()
    length = 3 ** N

    # Precompute powers of 3 up to N
    pow3 = [1] * (N + 1)
    for i in range(1, N + 1):
        pow3[i] = pow3[i - 1] * 3

    # dp[l][r][v] = min changes to make segment A[l..r] (0-indexed) collapse to value v (0 or 1)
    # We'll compute recursively with memoization
    from functools import lru_cache

    @lru_cache(None)
    def dfs(l, r, target):
        # segment length = 3^k for some k
        if l == r:
            return 0 if int(A[l]) == target else 1
        # split into 3 equal parts
        part_len = (r - l + 1) // 3
        m1 = l + part_len
        m2 = m1 + part_len
        # We need to make each part collapse to some value (0 or 1)
        # Then the majority of those three values must be target
        best = float('inf')
        # Enumerate possible triples (a,b,c) in {0,1}^3 whose majority is target
        for a in (0, 1):
            for b in (0, 1):
                for c in (0, 1):
                    if (a + b + c) >= 2:
                        majority = 1
                    else:
                        majority = 0
                    if majority != target:
                        continue
                    cost = dfs(l, m1 - 1, a) + dfs(m1, m2 - 1, b) + dfs(m2, r, c)
                    if cost < best:
                        best = cost
        return best

    # We want to change A'_1 from its current value to the opposite
    # First compute current A'_1 without changing anything
    # We can compute by recursively applying majority rule
    def get_current(l, r):
        if l == r:
            return int(A[l])
        part_len = (r - l + 1) // 3
        m1 = l + part_len
        m2 = m1 + part_len
        a = get_current(l, m1 - 1)
        b = get_current(m1, m2 - 1)
        c = get_current(m2, r)
        return 1 if (a + b + c) >= 2 else 0

    current = get_current(0, length - 1)
    target = 1 - current
    answer = dfs(0, length - 1, target)
    print(answer)

if __name__ == "__main__":
    solve()
