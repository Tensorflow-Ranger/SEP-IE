```python
import sys
import math

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    K = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    max_val = max(A)
    freq = [0] * (max_val + 1)
    for x in A:
        freq[x] += 1

    # prefix sums of frequencies for multiples
    prefix = [0] * (max_val + 2)
    for i in range(1, max_val + 1):
        cnt = 0
        for m in range(i, max_val + 1, i):
            cnt += freq[m]
        prefix[i] = cnt

    ans = [0] * N
    for idx, val in enumerate(A):
        # total count of numbers divisible by val
        total_divisible = prefix[val]
        if total_divisible < K:
            ans[idx] = 0
            continue

        # we need to check divisors of val
        best = 0
        # iterate over divisors of val
        d = 1
        while d * d <= val:
            if val % d == 0:
                div1 = d
                div2 = val // d
                if prefix[div1] >= K:
                    best = max(best, div1)
                if div2 != div1 and prefix[div2] >= K:
                    best = max(best, div2)
            d += 1
        ans[idx] = best

    sys.stdout.write("\n".join(map(str, ans)))

if __name__ == "__main__":
    main()
```
