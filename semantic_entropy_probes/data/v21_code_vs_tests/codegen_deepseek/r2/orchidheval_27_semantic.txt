def flip_case(string: str) -> str:
    """ For a given string, reverse the case of all characters.
    """
    result = []
    for ch in string:
        if ch.islower():
            result.append(ch.upper())
        elif ch.isupper():
            result.append(ch.lower())
        else:
            result.append(ch)
    return ''.join(result)
