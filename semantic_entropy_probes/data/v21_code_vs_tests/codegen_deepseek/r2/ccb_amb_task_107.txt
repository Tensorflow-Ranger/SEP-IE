```python
import sys
import math

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    K = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    max_val = max(A)
    freq = [0] * (max_val + 1)
    for x in A:
        freq[x] += 1

    # prefix sums of frequencies for multiples
    prefix = [0] * (max_val + 1)
    for d in range(1, max_val + 1):
        cnt = 0
        for m in range(d, max_val + 1, d):
            cnt += freq[m]
        prefix[d] = cnt

    # For each i, we need the largest d such that prefix[d] >= K
    # and d divides A[i].
    # Precompute the largest d for each possible value.
    best_for_val = [0] * (max_val + 1)
    for d in range(max_val, 0, -1):
        if best_for_val[d] == 0:
            best_for_val[d] = d
        # propagate to divisors
        for m in range(2 * d, max_val + 1, d):
            if best_for_val[m] == 0:
                best_for_val[m] = d

    out_lines = []
    for x in A:
        # We need the largest divisor d of x such that prefix[d] >= K
        ans = 1
        # iterate over divisors of x
        # We can use precomputed best_for_val to speed up, but we still need to check prefix condition.
        # Instead, we can iterate over divisors of x in descending order using sqrt factorization.
        divisors = []
        sqrt_x = int(math.isqrt(x))
        for d in range(1, sqrt_x + 1):
            if x % d == 0:
                divisors.append(d)
                if d != x // d:
                    divisors.append(x // d)
        divisors.sort(reverse=True)
        for d in divisors:
            if prefix[d] >= K:
                ans = d
                break
        out_lines.append(str(ans))

    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    main()
```
