```python
import sys

MOD = 998244353

def modinv(x):
    return pow(x, MOD - 2, MOD)

def solve():
    data = sys.stdin.read().split()
    it = iter(data)
    H = int(next(it))
    W = int(next(it))
    N = H * W
    A = [0] * N
    for i in range(N):
        A[i] = int(next(it))
    Q = int(next(it))
    sh = int(next(it)) - 1
    sw = int(next(it)) - 1

    # Precompute factorials and inverse factorials for binomial coefficients
    fact = [1] * (H + W + 5)
    inv_fact = [1] * (H + W + 5)
    for i in range(1, len(fact)):
        fact[i] = fact[i - 1] * i % MOD
    inv_fact[-1] = modinv(fact[-1])
    for i in range(len(fact) - 2, -1, -1):
        inv_fact[i] = inv_fact[i + 1] * (i + 1) % MOD

    def nCr(n, r):
        if r < 0 or r > n:
            return 0
        return fact[n] * inv_fact[r] % MOD * inv_fact[n - r] % MOD

    # dp[h][w] = number of ways to reach (h,w) from (1,1) times product of values along path
    # Actually we only need dp for the whole grid: dp[i] = ways to reach cell i (0-indexed row-major)
    # dp[0] = A[0]
    # For cell (r,c): dp = A[r,c] * (dp from left + dp from top)
    # But we can compute the whole dp in O(HW) each time? Too slow.
    # Instead, note that each path passes through exactly one cell in each anti-diagonal.
    # Let k = r + c (0-indexed). Then number of paths through (r,c) is C(k, r) * C(H+W-2 - k, H-1-r)
    # So contribution of cell (r,c) to total sum is A[r,c] * C(k, r) * C(H+W-2 - k, H-1-r)
    # Therefore total sum = sum_{r,c} A[r,c] * coeff[r,c]
    # We can precompute coeff[r,c] for each cell.
    coeff = [0] * N
    total = 0
    for r in range(H):
        for c in range(W):
            idx = r * W + c
            k = r + c
            ways = nCr(k, r) * nCr(H + W - 2 - k, H - 1 - r) % MOD
            coeff[idx] = ways
            total = (total + A[idx] * ways) % MOD

    out_lines = []
    pos = sh * W + sw
    for _ in range(Q):
        d = next(it)
        a = int(next(it))
        # Update total: subtract old contribution, add new contribution
        total = (total - A[pos] * coeff[pos]) % MOD
        A[pos] = a
        total = (total + A[pos] * coeff[pos]) % MOD
        # Move position
        if d == 'L':
            sw -= 1
        elif d == 'R':
            sw += 1
        elif d == 'U':
            sh -= 1
        else:  # 'D'
            sh += 1
        pos = sh * W + sw
        out_lines.append(str(total % MOD))

    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```
