import sys

def solve() -> None:
    it = iter(sys.stdin.read().split())
    H = int(next(it))
    W = int(next(it))
    rows = []
    for _ in range(H):
        rows.append(int(next(it), 2))

    # Precompute column masks
    col_masks = [1 << j for j in range(W)]

    # For each column subset, compute the effect on each row
    # We'll try all 2^W column flips (W <= 18)
    # For each column flip mask, we can compute the resulting row values
    # Then for each row we choose to flip it or not to minimize its popcount

    # Precompute popcount for all numbers up to 2^W - 1
    max_val = 1 << W
    popcount = [0] * max_val
    for i in range(1, max_val):
        popcount[i] = popcount[i >> 1] + (i & 1)

    best_total = H * W  # upper bound

    # Try all column flip masks
    for col_mask in range(1 << W):
        total = 0
        # For each row, compute the value after column flips
        for r in rows:
            flipped = r ^ col_mask
            cnt = popcount[flipped]
            total += min(cnt, W - cnt)  # flip row if beneficial
        if total < best_total:
            best_total = total

    print(best_total)

if __name__ == "__main__":
    solve()
