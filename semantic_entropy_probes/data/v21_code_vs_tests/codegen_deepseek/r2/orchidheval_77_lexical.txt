def iscube(a):
    '''
    Write a function that takes an integer a and returns True 
    if this integer has an integer root.
    '''
    if a < 0:
        root = round(-((-a) ** (1/3)))
        return root ** 3 == a
    elif a > 0:
        root = round(a ** (1/3))
        return root ** 3 == a
    else:
        return True
