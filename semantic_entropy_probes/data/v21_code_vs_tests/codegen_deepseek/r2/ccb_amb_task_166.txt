```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    # We need the shortest subarray with a repeated value.
    # For each value, track its last seen index.
    last_seen = {}
    min_len = N + 1  # larger than any possible answer

    for i, x in enumerate(A):
        if x in last_seen:
            # subarray from last_seen[x] to i (inclusive) has repeated x
            length = i - last_seen[x] + 1
            if length < min_len:
                min_len = length
        last_seen[x] = i

    if min_len == N + 1:
        print(-1)
    else:
        print(min_len)

if __name__ == "__main__":
    solve()
```
