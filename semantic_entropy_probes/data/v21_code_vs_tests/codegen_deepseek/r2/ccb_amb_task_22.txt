import sys
from bisect import bisect_right

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    W = int(next(it))
    blocks = []
    for _ in range(N):
        x = int(next(it))
        y = int(next(it))
        blocks.append((x, y))
    # group blocks by column
    by_col = [[] for _ in range(W + 1)]
    for idx, (x, y) in enumerate(blocks):
        by_col[x].append((y, idx))
    # sort each column by y ascending
    for col in range(1, W + 1):
        by_col[col].sort()
    # for each block, compute its removal time (or INF if never removed)
    INF = 10 ** 20
    removal_time = [INF] * N
    # process columns independently
    for col in range(1, W + 1):
        lst = by_col[col]
        if not lst:
            continue
        # stack of (y, idx) where y is increasing
        stack = []
        for y, idx in lst:
            # while stack not empty and top y == y-1 (adjacent downward)
            while stack and stack[-1][0] == y - 1:
                _, prev_idx = stack.pop()
                # removal time of prev_idx is y (the time when the lower one reaches bottom row)
                removal_time[prev_idx] = y
                y += 1  # after removal, the block above falls one step, so we check again
            stack.append((y, idx))
    # answer queries
    Q = int(next(it))
    out_lines = []
    for _ in range(Q):
        t = int(next(it))
        a = int(next(it)) - 1
        rt = removal_time[a]
        if rt <= t:
            out_lines.append("No")
        else:
            out_lines.append("Yes")
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
