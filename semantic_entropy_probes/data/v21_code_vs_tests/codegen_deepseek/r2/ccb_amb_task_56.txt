```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    arr = list(map(int, data[1:1+n]))
    arr.sort()

    max_val = arr[-1]
    count = [0] * (max_val + 1)
    for x in arr:
        count[x] = 1

    ans = 0
    for i in range(n):
        a = arr[i]
        # iterate over possible differences
        # For each a, we try b = a + d, c = a + 2d
        # Since a < b < c and all are in S, we can loop d such that a + 2d <= max_val
        max_d = (max_val - a) // 2
        for d in range(1, max_d + 1):
            b = a + d
            c = a + 2 * d
            if count[b] and count[c]:
                ans += 1
    print(ans)

if __name__ == "__main__":
    main()
```
