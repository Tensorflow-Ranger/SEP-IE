import sys

class Fenwick:
    def __init__(self, n):
        self.n = n
        self.bit = [0] * (n + 1)

    def update(self, idx, delta):
        i = idx + 1
        while i <= self.n:
            self.bit[i] += delta
            i += i & -i

    def query(self, idx):
        # sum of [0, idx]
        s = 0
        i = idx + 1
        while i > 0:
            s += self.bit[i]
            i -= i & -i
        return s

    def range_sum(self, l, r):
        # sum of [l, r]
        if l > r:
            return 0
        res = self.query(r)
        if l > 0:
            res -= self.query(l - 1)
        return res
