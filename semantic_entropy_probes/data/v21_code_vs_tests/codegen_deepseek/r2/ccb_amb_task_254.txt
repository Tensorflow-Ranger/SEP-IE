```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    A = list(map(int, data[1:1 + n]))

    # Count distinct elements from left and right
    left_distinct = [0] * (n + 1)
    right_distinct = [0] * (n + 1)

    seen = set()
    for i in range(n):
        seen.add(A[i])
        left_distinct[i + 1] = len(seen)

    seen.clear()
    for i in range(n - 1, -1, -1):
        seen.add(A[i])
        right_distinct[i] = len(seen)

    # For each split position i (1 <= i <= n-1)
    # left part: indices 0..i-1 -> length i -> left_distinct[i]
    # right part: indices i..n-1 -> right_distinct[i]
    ans = 0
    for i in range(1, n):
        total = left_distinct[i] + right_distinct[i]
        if total > ans:
            ans = total

    print(ans)

if __name__ == "__main__":
    solve()
```
