def words_string(s):
    """
    You will be given a string of words separated by standard delimiters. Your task is
    to split the string into words and return an array of the words.
    """
    import re
    if not s:
        return []
    # Split on commas and/or spaces, treating consecutive delimiters as one
    return re.split(r'[,\s]+', s.strip())
