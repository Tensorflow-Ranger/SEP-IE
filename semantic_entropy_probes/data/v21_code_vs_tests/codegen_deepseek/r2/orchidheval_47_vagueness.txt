def median(l: list):
    """Return the median of the list l."""
    if not l:
        raise ValueError("List must not be empty")
    sorted_l = sorted(l)
    n = len(sorted_l)
    mid = n // 2
    if n % 2 == 1:
        return float(sorted_l[mid])
    else:
        return (sorted_l[mid - 1] + sorted_l[mid]) / 2.0
