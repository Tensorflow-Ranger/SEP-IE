from collections import defaultdict

def solve() -> None:

    n = int(next(it))
    a = [int(next(it)) for _ in range(n)]

    total_cnt = defaultdict(int)

        total_cnt[val] += 1

    left_size = 0

    # but it's easier to handle inside loops
