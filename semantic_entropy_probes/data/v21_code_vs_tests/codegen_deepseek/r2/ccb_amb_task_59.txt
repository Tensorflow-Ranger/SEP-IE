```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    arr = list(map(int, data[1:1+n]))
    arr.sort()
    
    max_val = arr[-1]
    count = [0] * (max_val + 1)
    for v in arr:
        count[v] = 1
    
    total = 0
    for i in range(n):
        a = arr[i]
        # iterate possible differences
        # For each b = a + d, c = a + 2d must be <= max_val
        # d must be positive
        max_d = (max_val - a) // 2
        for d in range(1, max_d + 1):
            b = a + d
            c = a + 2 * d
            if count[b] and count[c]:
                total += 1
    print(total)

if __name__ == "__main__":
    main()
```
