```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    # We need the shortest subarray with a repeated value.
    # That means two equal numbers with minimal distance between them.
    # The length of such a subarray is (j - i + 1) where i and j are indices with A[i] == A[j].
    # So we track the last seen position of each value.
    last_pos = {}
    min_len = N + 1  # larger than any possible

    for idx, val in enumerate(A):
        if val in last_pos:
            dist = idx - last_pos[val] + 1
            if dist < min_len:
                min_len = dist
        last_pos[val] = idx

    if min_len == N + 1:
        print(-1)
    else:
        print(min_len)

if __name__ == "__main__":
    solve()
```
