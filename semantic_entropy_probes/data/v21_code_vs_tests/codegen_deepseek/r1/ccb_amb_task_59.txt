```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    arr = list(map(int, data[1:1 + n]))
    arr.sort()

    max_val = arr[-1]
    count = [0] * (max_val + 1)
    for x in arr:
        count[x] = 1

    ans = 0
    for i in range(n):
        a = arr[i]
        max_step = (max_val - a) // 2
        for step in range(1, max_step + 1):
            b = a + step
            c = a + 2 * step
            if c <= max_val and count[b] and count[c]:
                ans += 1
    print(ans)

if __name__ == "__main__":
    main()
```
