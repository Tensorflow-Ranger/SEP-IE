```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    arr = [int(next(it)) for _ in range(n)]

    # Count distinct elements from left to right
    seen_left = set()
    distinct_left = [0] * (n - 1)
    for i in range(n - 1):
        seen_left.add(arr[i])
        distinct_left[i] = len(seen_left)

    # Count distinct elements from right to left
    seen_right = set()
    distinct_right = [0] * (n - 1)
    for i in range(n - 1, 0, -1):
        seen_right.add(arr[i])
        distinct_right[i - 1] = len(seen_right)

    # Compute maximum sum
    ans = 0
    for i in range(n - 1):
        ans = max(ans, distinct_left[i] + distinct_right[i])
    print(ans)

if __name__ == "__main__":
    solve()
```
