```python
import sys

MOD = 998244353

def modinv(x):
    return pow(x, MOD - 2, MOD)

def main():
    input = sys.stdin.readline
    H, W = map(int, input().split())
    grid = [list(map(int, input().split())) for _ in range(H)]
    Q, sh, sw = map(int, input().split())
    sh -= 1
    sw -= 1

    dp = [[0] * W for _ in range(H)]
    dp[0][0] = grid[0][0]
    for i in range(H):
        for j in range(W):
            if i == 0 and j == 0:
                continue
            val = 0
            if i > 0:
                val += dp[i-1][j]
            if j > 0:
                val += dp[i][j-1]
            dp[i][j] = val * grid[i][j] % MOD

    ans = dp[H-1][W-1]
    out_lines = []

    for _ in range(Q):
        d, a = input().split()
        a = int(a)
        h, w = sh, sw
        if d == 'L':
            w -= 1
        elif d == 'R':
            w += 1
        elif d == 'U':
            h -= 1
        else:  # 'D'
            h += 1
        sh, sw = h, w

        old = grid[h][w]
        if old == a:
            out_lines.append(str(ans))
            continue

        inv_old = modinv(old)
        factor = a * inv_old % MOD

        dp[h][w] = dp[h][w] * factor % MOD
        grid[h][w] = a

        i, j = h, w
        while i < H and j < W:
            if i > 0 or j > 0:
                val = 0
                if i > 0:
                    val += dp[i-1][j]
                if j > 0:
                    val += dp[i][j-1]
                dp[i][j] = val * grid[i][j] % MOD
            i += 1
            j += 1

        ans = dp[H-1][W-1]
        out_lines.append(str(ans))

    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    main()
```
