```python
import sys
import math

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    K = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    max_val = max(A)
    freq = [0] * (max_val + 1)
    for x in A:
        freq[x] += 1

    # Count multiples for each possible divisor
    multiples = [0] * (max_val + 1)
    for d in range(1, max_val + 1):
        for m in range(d, max_val + 1, d):
            multiples[d] += freq[m]

    # Precompute gcds for each i
    ans = [0] * N
    for i, val in enumerate(A):
        # We need K elements including A[i], so we need at least K-1 other elements
        # with a common divisor d that also divides val.
        # For each divisor d of val, check if multiples[d] >= K.
        # The largest such d is the answer for i.
        best = 0
        # Iterate over divisors of val efficiently
        limit = int(math.isqrt(val))
        for d1 in range(1, limit + 1):
            if val % d1 == 0:
                d2 = val // d1
                if multiples[d1] >= K:
                    best = max(best, d1)
                if d2 != d1 and multiples[d2] >= K:
                    best = max(best, d2)
        ans[i] = best

    sys.stdout.write("\n".join(map(str, ans)) + "\n")

if __name__ == "__main__":
    main()
```
