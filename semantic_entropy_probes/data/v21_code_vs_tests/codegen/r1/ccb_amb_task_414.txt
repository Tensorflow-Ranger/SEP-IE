import sys
import math
input = sys.stdin.read

MOD = 998244353

def main():
    data = input().split()
    idx = 0
    H = int(data[idx]); idx +=1
    W = int(data[idx]); idx +=1
    A = []
    for _ in range(H):
        row = list(map(int, data[idx:idx+W]))
        idx += W
        A.append(row)
    Q = int(data[idx]); idx +=1
    sh = int(data[idx]); idx +=1
    sw = int(data[idx]); idx +=1

    # Initialize the grid
    grid = [row[:] for row in A]

    # Precompute factorials and inverse factorials for combinations
    max_n = H + W
    fact = [1] * (max_n + 1)
    inv_fact = [1] * (max_n + 1)
    for i in range(1, max_n + 1):
        fact[i] = fact[i-1] * i % MOD
    inv_fact[max_n] = pow(fact[max_n], MOD-2, MOD)
    for i in range(max_n-1, -1, -1):
        inv_fact[i] = inv_fact[i+1] * (i+1) % MOD

    def comb(n, k):
        if k < 0 or k > n:
            return 0
        return fact[n] * inv_fact[k] % MOD * inv_fact[n - k] % MOD

    # Precompute the number of paths from (1,1) to (H,W)
    total_paths = comb(H + W - 2, H - 1)

    # Initialize dp table
    dp = [[0] * (W + 1) for _ in range(H + 1)]
    dp[1][1] = 1

    for _ in range(Q):
        d = data[idx]; idx +=1
        a = int(data[idx]); idx +=1

        # Move Takahashi
        h, w = sh, sw
        if d == 'L':
            w -= 1
        elif d == 'R':
            w += 1
        elif d == 'U':
            h -= 1
        elif d == 'D':
            h += 1
        # Update the grid
        grid[h-1][w-1] = a
        sh, sw = h, w

        # Rebuild dp table
        dp = [[0] * (W + 1) for _ in range(H + 1)]
        dp[1][1] = 1
        for i in range(1, H + 1):
            for j in range(1, W + 1):
                if i == 1 and j == 1:
                    continue
                val = grid[i-1][j-1]
                if i > 1:
                    dp[i][j] = dp[i-1][j] * val % MOD
                if j > 1:
                    dp[i][j] = (dp[i][j] + dp[i][j-1] * val) % MOD

        # The answer is the value at (H, W)
        print(dp[H][W] % MOD)

if __name__ == "__main__":
    main()
