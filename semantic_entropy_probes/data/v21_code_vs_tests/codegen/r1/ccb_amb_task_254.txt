```python
import sys
import math

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    N = int(data[0])
    A = list(map(int, data[1:]))

    prefix = [0] * (N + 1)
    suffix = [0] * (N + 1)
    seen = set()
    count = 0

    # Compute prefix distinct counts
    for i in range(N):
        if A[i] not in seen:
            seen.add(A[i])
            count += 1
        prefix[i + 1] = count
        seen = set()
        count = 0

    seen = set()
    count = 0

    # Compute suffix distinct counts
    for i in range(N - 1, -1, -1):
        if A[i] not in seen:
            seen.add(A[i])
            count += 1
        suffix[i] = count
        seen = set()
        count = 0

    max_sum = 0
    for i in range(1, N):
        current = prefix[i] + suffix[i]
        if current > max_sum:
            max_sum = current

    print(max_sum)

if __name__ == "__main__":
    main()
```
