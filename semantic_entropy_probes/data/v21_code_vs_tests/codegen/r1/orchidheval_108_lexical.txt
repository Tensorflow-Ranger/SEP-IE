def count_nums(arr):
    import math
    count = 0
    for num in arr:
        digits_sum = 0
        if num < 0:
            num = -num
            first_digit = -int(str(num)[0])
            digits_sum += first_digit
            num = num // 10
        while num > 0:
            digits_sum += num % 10
            num = num // 10
        if digits_sum > 0:
            count += 1
    return count
