```python
from collections import Counter

def main():
    import sys
    input = sys.stdin.read
    cards = list(map(int, input().split()))
    count = Counter(cards)
    values = list(count.values())
    if sorted(values) == [2, 3]:
        print("Yes")
    else:
        print("No")

if __name__ == "__main__":
    main()
```
