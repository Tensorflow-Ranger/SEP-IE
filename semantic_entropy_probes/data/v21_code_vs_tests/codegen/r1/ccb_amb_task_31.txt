```python
import sys
import math

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    N = int(data[0])
    A = data[1]

    length = 3 ** N
    target = '0' if A.count('1') < length // 2 else '1'

    def apply_operation(s):
        n = len(s)
        result = []
        for i in range(0, n, 3):
            group = s[i:i+3]
            count_0 = group.count('0')
            count_1 = group.count('1')
            if count_0 > count_1:
                result.append('0')
            else:
                result.append('1')
        return ''.join(result)

    current = A
    for _ in range(N):
        current = apply_operation(current)

    if current == target:
        print(0)
    else:
        print(1)

if __name__ == "__main__":
    main()
```
