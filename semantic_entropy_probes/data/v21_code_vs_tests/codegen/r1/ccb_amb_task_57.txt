```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    N = int(data[0])
    S = list(map(int, data[1:N+1]))
    S.sort()
    count = 0
    for i in range(N):
        for j in range(i+1, N):
            a = S[i]
            b = S[j]
            c = 2 * b - a
            if c in S:
                count += 1
    print(count)

if __name__ == "__main__":
    main()
```
