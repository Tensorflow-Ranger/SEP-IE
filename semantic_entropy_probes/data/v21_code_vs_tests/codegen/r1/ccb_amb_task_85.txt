```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    
    N = int(data[0])
    P = list(map(int, data[1:N+1]))
    Q = list(map(int, data[N+1:2*N+1]))
    
    # Create a mapping from bib number to person index
    bib_to_index = {bib: i for i, bib in enumerate(Q)}
    
    # For each bib number i, find the person staring at it (P[i]-1)
    # Then get the bib number of that person
    result = [0] * N
    for i in range(N):
        person_staring_at_i = P[i] - 1
        bib_of_that_person = Q[person_staring_at_i]
        result[i] = bib_of_that_person
    
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
```
