def digits(n):
    """Given a positive integer n, return the product of the key digits.
    Return 0 if all digits are even.
    """
    product = 1
    has_odd = False
    for digit in str(n):
        d = int(digit)
        if d % 2 == 1:
            product *= d
            has_odd = True
    return product if has_odd else 0
