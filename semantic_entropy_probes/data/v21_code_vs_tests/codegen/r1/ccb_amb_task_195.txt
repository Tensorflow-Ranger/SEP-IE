```python
import sys
import itertools

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    H = int(data[0])
    W = int(data[1])
    grid = data[2:2+H]
    
    # Convert each row to a list of integers
    grid = [list(map(int, list(row))) for row in grid]
    
    # Precompute the sum of each row
    row_sums = [sum(row) for row in grid]
    
    # Try all possible combinations of flipping columns (since W <= 18)
    min_total = float('inf')
    for mask in itertools.product([0, 1], repeat=W):
        total = 0
        for i in range(H):
            row_sum = row_sums[i]
            flip_count = sum(mask)
            # If the number of flips is even, the row sum remains the same
            # If the number of flips is odd, the row sum becomes W - row_sum
            if flip_count % 2 == 1:
                total += (W - row_sum)
            else:
                total += row_sum
        min_total = min(min_total, total)
    
    print(min_total)

if __name__ == "__main__":
    main()
```
