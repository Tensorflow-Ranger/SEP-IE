```python
import sys
import heapq
from collections import defaultdict

def main():
    input = sys.stdin.read
    data = input().split()
    idx = 0

    N = int(data[idx])
    idx += 1

    W = list(map(int, data[idx:idx+N]))
    idx += N

    L = []
    R = []
    for _ in range(N):
        L.append(int(data[idx]))
        R.append(int(data[idx+1]))
        idx += 2

    Q = int(data[idx])
    idx += 1

    queries = []
    for _ in range(Q):
        s = int(data[idx])
        t = int(data[idx+1])
        queries.append((s-1, t-1))
        idx += 2

    # Build adjacency list for intervals
    adj = [[] for _ in range(N)]
    for i in range(N):
        for j in range(i+1, N):
            if R[i] >= L[j] and R[j] >= L[i]:
                adj[i].append(j)
                adj[j].append(i)

    # Dijkstra's algorithm to find minimum weight path
    import heapq
    INF = float('inf')
    dist = [INF] * N
    dist[0] = W[0]
    heap = [(W[0], 0)]

    while heap:
        weight, u = heapq.heappop(heap)
        if weight > dist[u]:
            continue
        for v in adj[u]:
            if dist[v] > dist[u] + W[v]:
                dist[v] = dist[u] + W[v]
                heapq.heappush(heap, (dist[v], v))

    # Process queries
    for s, t in queries:
        if dist[t] == INF:
            print(-1)
        else:
            print(dist[t])

if __name__ == "__main__":
    main()
```
