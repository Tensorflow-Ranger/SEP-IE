```python
import sys
import math

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    N = int(data[0])
    A = data[1]

    length = 3 ** N
    target = '0' if A[0] == '1' else '1'

    def apply_operation(s):
        n = len(s)
        result = []
        for i in range(0, n, 3):
            group = s[i:i+3]
            count0 = group.count('0')
            count1 = group.count('1')
            if count0 > count1:
                result.append('0')
            else:
                result.append('1')
        return ''.join(result)

    current = A
    for _ in range(N):
        current = apply_operation(current)

    if current[0] == target:
        print(0)
    else:
        min_changes = 0
        for i in range(3 ** N):
            if A[i] != target:
                min_changes += 1
        print(min_changes)

if __name__ == "__main__":
    main()
```
