```python
import sys
import bisect

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    N = int(data[idx])
    idx += 1
    W = int(data[idx])
    idx += 1
    
    pos = []
    for _ in range(N):
        x = int(data[idx])
        idx += 1
        y = int(data[idx])
        idx += 1
        pos.append((x, y))
    
    Q = int(data[idx])
    idx += 1
    
    queries = []
    for _ in range(Q):
        t = int(data[idx])
        idx += 1
        a = int(data[idx]) - 1
        idx += 1
        queries.append((t, a))
    
    # Sort the positions by Y coordinate
    pos.sort(key=lambda x: x[1])
    
    # For each column, keep track of the Y coordinates of blocks
    from collections import defaultdict
    col_y = defaultdict(list)
    for x, y in pos:
        bisect.insort(col_y[x], y)
    
    # For each column, we can compute the number of blocks that would fall out
    # after t steps using binary search
    
    # Precompute for each column the sorted list of Y coordinates
    # and the prefix sums for binary search
    # We will process queries offline by sorting them by time
    
    # Sort queries by time
    queries.sort()
    
    # For each column, precompute the sorted list of Y coordinates
    # and the prefix sums for binary search
    col_data = []
    for x in range(1, W + 1):
        ys = col_y[x]
        if not ys:
            col_data.append((x, ys, []))
            continue
        # For each Y in ys, compute the number of blocks that would fall out
        # after t steps, using binary search
        # We can precompute the prefix sums for binary search
        # For a given t, the number of blocks that fall out is the number of Y's
        # such that Y + t >= the number of blocks below it
        # This is a bit tricky, but we can use binary search for each query
        col_data.append((x, ys, []))
    
    # Process queries
    results = []
    for t, a in queries:
        x, y = pos[a]
        # For the block at (x, y), compute how many steps it would take to fall out
        # We need to find the number of blocks in column x with Y <= y
        # and how many steps it would take for this block to fall out
        # The block will fall out after t steps if it is not removed before
        # To compute this, we can find the number of blocks in column x with Y < y
        # and the number of blocks in column x with Y <= y
        # The number of blocks that fall out in t steps is the number of blocks
        # in column x that are in the first (t + 1) positions
        ys = col_data[x - 1][1]
        # Find the number of blocks with Y <= y
        cnt = bisect.bisect_right(ys, y)
        if cnt <= t:
            results.append("No")
        else:
            results.append("Yes")
    
    print("\n".join(results))

if __name__ == "__main__":
    main()
```
