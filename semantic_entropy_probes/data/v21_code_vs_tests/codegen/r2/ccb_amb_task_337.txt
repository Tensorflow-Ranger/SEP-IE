```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    N = int(data[0])
    P = list(map(int, data[1:]))
    
    pos = [0] * (N + 1)
    for i in range(N):
        pos[P[i]] = i
    
    total_cost = 0
    for i in range(N, 0, -1):
        current = pos[i]
        if current == i - 1:
            continue
        total_cost += current
        pos[P[current]] = current
        pos[i] = current
        P[current], P[current + 1] = P[current + 1], P[current]
    
    print(total_cost)

if __name__ == "__main__":
    main()
```
