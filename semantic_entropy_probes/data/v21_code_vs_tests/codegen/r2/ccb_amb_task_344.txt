```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    
    N = int(data[0])
    A = list(map(int, data[1:N+1]))
    B = list(map(int, data[N+1:2*N+1]))
    C = list(map(int, data[2*N+1:3*N+1]))
    
    cost = 0
    for a, b, c in zip(A, B, C):
        if a != b:
            cost += c
    print(cost)

if __name__ == "__main__":
    main()
```
