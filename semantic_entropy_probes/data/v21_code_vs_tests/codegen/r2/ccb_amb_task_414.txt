```python
import sys
input = sys.stdin.read
MOD = 998244353

def main():
    data = input().split()
    idx = 0
    H = int(data[idx]); idx +=1
    W = int(data[idx]); idx +=1
    A = [[0]*(W+1) for _ in range(H+1)]
    for h in range(1, H+1):
        for w in range(1, W+1):
            A[h][w] = int(data[idx]); idx +=1
    Q = int(data[idx]); idx +=1
    sh = int(data[idx]); idx +=1
    sw = int(data[idx]); idx +=1

    # dp[i][j] = sum of products of paths from (1,1) to (i,j)
    dp = [[0]*(W+1) for _ in range(H+1)]
    dp[1][1] = 1

    for h in range(1, H+1):
        for w in range(1, W+1):
            if h == 1 and w == 1:
                continue
            val = 0
            if h > 1:
                val = (val + dp[h-1][w]) % MOD
            if w > 1:
                val = (val + dp[h][w-1]) % MOD
            dp[h][w] = val

    for _ in range(Q):
        d = data[idx]; idx +=1
        a = int(data[idx]); idx +=1

        # Move
        if d == 'L':
            sw -= 1
        elif d == 'R':
            sw += 1
        elif d == 'U':
            sh -= 1
        elif d == 'D':
            sh += 1

        # Update the cell
        h, w = sh, sw
        A[h][w] = a

        # Recompute dp
        dp = [[0]*(W+1) for _ in range(H+1)]
        dp[1][1] = 1
        for h in range(1, H+1):
            for w in range(1, W+1):
                if h == 1 and w == 1:
                    continue
                val = 0
                if h > 1:
                    val = (val + dp[h-1][w]) % MOD
                if w > 1:
                    val = (val + dp[h][w-1]) % MOD
                dp[h][w] = val
        print(dp[H][W] % MOD)

if __name__ == "__main__":
    main()
```
