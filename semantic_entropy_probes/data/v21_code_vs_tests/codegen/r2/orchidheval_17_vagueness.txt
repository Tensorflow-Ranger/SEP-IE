from typing import List

def parse_music(music_string: str) -> List[int]:
    tokens = music_string.split()
    result = []
    for token in tokens:
        if token == 'o':
            result.append(1)
        elif token == 'o|':
            result.append(2)
        elif token == '.|':
            result.append(4)
    return result
