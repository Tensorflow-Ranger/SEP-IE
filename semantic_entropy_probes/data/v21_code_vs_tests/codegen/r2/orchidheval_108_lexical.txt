def count_nums(arr):
    import math
    count = 0
    for num in arr:
        digits = []
        if num < 0:
            num = -num
            digits.append(-1)
        else:
            digits.append(1)
        while num > 0:
            digits.append(num % 10)
            num = num // 10
        total = sum(digits)
        if total > 0:
            count += 1
    return count
