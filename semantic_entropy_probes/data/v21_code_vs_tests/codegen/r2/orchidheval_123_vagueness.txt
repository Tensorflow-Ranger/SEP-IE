def get_odd_collatz(n):
    import math
    if not isinstance(n, int) or n <= 0:
        return []
    collatz = []
    current = n
    while current != 1:
        if current % 2 == 1:
            collatz.append(current)
        if current % 2 == 0:
            current = current // 2
        else:
            current = 3 * current + 1
    if current % 2 == 1:
        collatz.append(current)
    return sorted(collatz)
