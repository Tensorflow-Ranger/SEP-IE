```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    p = list(map(int, data[1:]))

    pos = [0] * (n + 1)
    for i, val in enumerate(p):
        pos[val] = i  # position of value val in p (0‑based)

    total_cost = 0
    i = 1
    while i <= n:
        if pos[i] == i - 1:
            i += 1
            continue
        # find the length of consecutive decreasing segment starting at i
        start = i
        while i + 1 <= n and pos[i + 1] == pos[i] - 1:
            i += 1
        end = i
        length = end - start + 1
        # cost for this segment is sum_{k=start}^{end} (k-1)
        # which equals (start-1 + end-1) * length // 2
        total_cost += (start - 1 + end - 1) * length // 2
        i += 1

    print(total_cost)

if __name__ == "__main__":
    solve()
```
