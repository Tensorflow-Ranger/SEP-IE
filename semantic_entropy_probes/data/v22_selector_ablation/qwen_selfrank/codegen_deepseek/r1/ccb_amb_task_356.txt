import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    N = int(next(it))
    W = [0] + [int(next(it)) for _ in range(N)]
    intervals = [(0, 0)]  # dummy for 1‑based indexing
    for _ in range(N):
        L = int(next(it))
        R = int(next(it))
        intervals.append((L, R))

    # Build complement graph connectivity via interval union‑find
    # Sort intervals by left endpoint
    order = list(range(1, N + 1))
    order.sort(key=lambda i: intervals[i][0])

    parent = list(range(N + 1))
    size = [1] * (N + 1)

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra == rb:
            return
        if size[ra] < size[rb]:
            ra, rb = rb, ra
        parent[rb] = ra
        size[ra] += size[rb]

    # Sweep line to union overlapping intervals
    max_r = -1
    last_idx = -1
    for idx in order:
        L, R = intervals[idx]
        if L <= max_r + 1:
            # overlaps or touches
            union(last_idx, idx)
            max_r = max(max_r, R)
        else:
            max_r = R
        last_idx = idx

    # Component mapping and component weight preprocessing
    comp_id = [0] * (N + 1)
    comp_min_weight = {}
    for i in range(1, N + 1):
        r = find(i)
        comp_id[i] = r
        if r not in comp_min_weight:
            comp_min_weight[r] = W[i]
        else:
            comp_min_weight[r] = min(comp_min_weight[r], W[i])

    # Process queries
    Q = int(next(it))
    out_lines = []
    for _ in range(Q):
        s = int(next(it))
        t = int(next(it))
        if comp_id[s] != comp_id[t]:
            out_lines.append("-1")
        else:
            # Path weight = W[s] + W[t] + (min weight in component except possibly endpoints)
            comp_root = comp_id[s]
            min_in_comp = comp_min_weight[comp_root]
            ans = W[s] + W[t]
            if min_in_comp != W[s] and min_in_comp != W[t]:
                ans += min_in_comp
            # If component has only two vertices, edge exists directly
            # but formula still works because min_in_comp is one of them.
            out_lines.append(str(ans))

    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
