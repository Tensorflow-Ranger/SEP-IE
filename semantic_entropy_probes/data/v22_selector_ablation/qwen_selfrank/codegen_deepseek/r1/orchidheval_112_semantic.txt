def reverse_delete(s, c):
    # Build result string by including only characters not in c
    result_chars = [ch for ch in s if ch not in c]
    result_str = ''.join(result_chars)
    # Check if result string is palindrome
    is_palindrome = result_str == result_str[::-1]
    return (result_str, is_palindrome)
