```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    t = int(next(it))
    out_lines = []
    for _ in range(t):
        n = int(next(it))
        A = [int(next(it)) for _ in range(2 * n)]
        pos = [[] for _ in range(n + 1)]
        for idx, val in enumerate(A):
            pos[val].append(idx)
        
        # adjacency check and compute "type"
        type_list = [0] * (n + 1)  # 0: adjacent, 1: separated and even distance, 2: separated and odd distance
        for x in range(1, n + 1):
            p1, p2 = pos[x]
            if abs(p1 - p2) == 1:
                type_list[x] = 0
            else:
                # distance between occurrences
                dist = abs(p1 - p2) - 1
                if dist % 2 == 0:
                    type_list[x] = 1
                else:
                    type_list[x] = 2
        
        # Count couples of type 1 and type 2
        cnt1 = 0
        cnt2 = 0
        for x in range(1, n + 1):
            if type_list[x] == 1:
                cnt1 += 1
            elif type_list[x] == 2:
                cnt2 += 1
        
        # Answer: choose two from type 1 + choose two from type 2
        ans = cnt1 * (cnt1 - 1) // 2 + cnt2 * (cnt2 - 1) // 2
        out_lines.append(str(ans))
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```
