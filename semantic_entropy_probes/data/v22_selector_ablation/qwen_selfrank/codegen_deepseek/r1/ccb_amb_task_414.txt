import sys

MOD = 998244353

def modinv(a):
    return pow(a, MOD - 2, MOD)

def solve():
    input = sys.stdin.readline
    H, W = map(int, input().split())
    A = [list(map(int, input().split())) for _ in range(H)]
    Q, sh, sw = map(int, input().split())
    sh -= 1
    sw -= 1

    # Precompute factorials and inverse factorials
    N = H + W + 5
    fact = [1] * (N)
    inv_fact = [1] * (N)
    for i in range(1, N):
        fact[i] = fact[i-1] * i % MOD
    inv_fact[N-1] = modinv(fact[N-1])
    for i in range(N-2, -1, -1):
        inv_fact[i] = inv_fact[i+1] * (i+1) % MOD

    def nCr(n, r):
        if r < 0 or r > n:
            return 0
        return fact[n] * inv_fact[r] % MOD * inv_fact[n-r] % MOD

    # Precompute path counts for each cell
    # path_to[h][w] = number of ways from (1,1) to (h,w)
    # path_from[h][w] = number of ways from (h,w) to (H,W)
    path_to = [[0]*W for _ in range(H)]
    path_from = [[0]*W for _ in range(H)]

    for h in range(H):
        for w in range(W):
            if h == 0 and w == 0:
                path_to[h][w] = 1
            else:
                ways = 0
                if h > 0:
                    ways += path_to[h-1][w]
                if w > 0:
                    ways += path_to[h][w-1]
                path_to[h][w] = ways % MOD

    for h in range(H-1, -1, -1):
        for w in range(W-1, -1, -1):
            if h == H-1 and w == W-1:
                path_from[h][w] = 1
            else:
                ways = 0
                if h < H-1:
                    ways += path_from[h+1][w]
                if w < W-1:
                    ways += path_from[h][w-1]
                path_from[h][w] = ways % MOD

    # Compute initial total sum
    total = 0
    for h in range(H):
        for w in range(W):
            if A[h][w] != 0:
                ways = path_to[h][w] * path_from[h][w] % MOD
                total = (total + ways * A[h][w]) % MOD

    out_lines = []
    directions = {'L': (0, -1), 'R': (0, 1), 'U': (-1, 0), 'D': (1, 0)}

    for _ in range(Q):
        d, a = input().split()
        a = int(a)
        dh, dw = directions[d]
        nh, nw = sh + dh, sw + dw

        # Remove old contribution of (sh, sw)
        ways_old = path_to[sh][sw] * path_from[sh][sw] % MOD
        total = (total - ways_old * A[sh][sw]) % MOD

        # Update grid
        A[sh][sw] = a

        # Add new contribution of (sh, sw)
        total = (total + ways_old * A[sh][sw]) % MOD

        # Move to new cell
        sh, sw = nh, nw

        out_lines.append(str(total % MOD))

    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
