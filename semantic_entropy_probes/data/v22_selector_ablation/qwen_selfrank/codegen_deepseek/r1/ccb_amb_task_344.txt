```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    A = [int(next(it)) for _ in range(N)]
    B = [int(next(it)) for _ in range(N)]
    C = [int(next(it)) for _ in range(N)]

    total_cost = 0
    for i in range(N):
        if A[i] != B[i]:
            total_cost += sum(A[k] * C[k] for k in range(N))
            A[i] ^= 1  # flip A[i]

    print(total_cost)

if __name__ == "__main__":
    solve()
```
