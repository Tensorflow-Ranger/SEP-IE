import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    # total distinct in whole array
    total_set = set(A)
    total_distinct = len(total_set)

    left_seen = set()
    left_distinct = [0] * (N - 1)  # left_distinct[i] = distinct in A[0..i]
    for i in range(N - 1):
        left_seen.add(A[i])
        left_distinct[i] = len(left_seen)

    right_seen = set()
    right_distinct = [0] * (N - 1)  # right_distinct[i] = distinct in A[i+1..N-1]
    for i in range(N - 2, -1, -1):
        right_seen.add(A[i + 1])
        right_distinct[i] = len(right_seen)

    ans = 0
    for i in range(N - 1):
        ans = max(ans, left_distinct[i] + right_distinct[i])
    print(ans)

if __name__ == "__main__":
    solve()
