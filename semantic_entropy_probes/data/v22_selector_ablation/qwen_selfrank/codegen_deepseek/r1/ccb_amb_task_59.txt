```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    arr = list(map(int, data[1:1 + n]))
    max_val = max(arr)
    
    present = [0] * (max_val + 1)
    for x in arr:
        present[x] = 1
    
    # prefix sums to answer "how many numbers <= i are in S"
    prefix = [0] * (max_val + 1)
    for i in range(1, max_val + 1):
        prefix[i] = prefix[i - 1] + present[i]
    
    ans = 0
    # iterate over possible B
    for b in arr:
        # iterate over possible step d = B - A = C - B > 0
        # A = B - d, C = B + d
        max_d = min(b - 1, max_val - b)
        for d in range(1, max_d + 1):
            a = b - d
            c = b + d
            if present[a] and present[c]:
                ans += 1
    print(ans)

if __name__ == "__main__":
    solve()
```
