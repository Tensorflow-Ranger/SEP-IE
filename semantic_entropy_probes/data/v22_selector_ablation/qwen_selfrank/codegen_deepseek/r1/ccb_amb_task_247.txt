import sys

def solve() -> None:
    it = iter(sys.stdin.read().split())
    n = int(next(it))
    arr = [int(next(it)) for _ in range(n)]

    # Step 1: compute prefix distinct counts
    prefix_distinct = [0] * n
    seen = set()
    for i in range(n):
        seen.add(arr[i])
        prefix_distinct[i] = len(seen)

    # Step 2: compute suffix distinct counts
    suffix_distinct = [0] * n
    seen.clear()
    for i in range(n - 1, -1, -1):
        seen.add(arr[i])
        suffix_distinct[i] = len(seen)

    # Step 3: compute max prefix up to each position
    max_prefix = [0] * n
    max_prefix[0] = prefix_distinct[0]
    for i in range(1, n):
        max_prefix[i] = max(max_prefix[i - 1], prefix_distinct[i])

    # Step 4: compute max suffix from each position
    max_suffix = [0] * n
    max_suffix[n - 1] = suffix_distinct[n - 1]
    for i in range(n - 2, -1, -1):
        max_suffix[i] = max(max_suffix[i + 1], suffix_distinct[i])

    # But better: for each middle start ms, we can compute distinct in middle

    # some limit? Not directly.

        # Actually simpler: we can maintain a segment tree of max_prefix[ms-1] + distinct(ms, me) over ms.

        # We need a data structure that supports range add and range max. Use lazy segment tree.

    # Given constraints, we need O(n) or O(n log n). Let's implement lazy segment tree.

    class LazySegTree:
        def __init__(self, n):
            self.size = 1
            while self.size < n:
                self.size <<= 1
            self.data = [0] * (2 * self.size)
            self.lazy = [0] * (2 * self.size)

        def _push(self, idx):
            if self.lazy[idx] != 0:
                self.data[2 * idx] += self.lazy[idx]
                self.data[2 * idx + 1] += self.lazy[idx]
                self.lazy[2 * idx] += self.lazy[idx]
                self.lazy[2 * idx + 1] += self.lazy[idx]
                self.lazy[idx] = 0

        def _update(self, l, r, v, idx, seg_l, seg_r):
            if r <= seg_l or seg_r <= l:
                return
            if l <= seg_l and seg_r <= r:
                self.data[idx] += v
                self.lazy[idx] += v
                return
            self._push(idx)
            mid = (seg_l + seg_r) // 2
            self._update(l, r, v, 2 * idx, seg_l, mid)
            self._update(l, r, v, 2 * idx + 1, mid, seg_r)
            self.data[idx] = max(self.data[2 * idx], self.data[2 * idx + 1])

        def update(self, l, r, v):
            self._update(l, r, v, 1, 0, self.size)

        def query(self, l, r, idx, seg_l, seg_r):
            if r <= seg_l or seg_r <= l:
                return -10**9
            if l <= seg_l and seg_r <= r:
                return self.data[idx]
            self._push(idx)
            mid = (seg_l + seg_r) // 2
            left = self.query(l, r, 2 * idx, seg_l, mid)
            right = self.query(l, r, 2 * idx + 1, mid, seg_r)
            return max(left, right)

        def range_max(self, l, r):
            return self.query(l, r, 1, 0, self.size)
