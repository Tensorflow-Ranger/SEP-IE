```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    last_seen = {}
    min_len = N + 1

    for i, val in enumerate(A):
        if val in last_seen:
            length = i - last_seen[val] + 1
            if length < min_len:
                min_len = length
        last_seen[val] = i

    if min_len == N + 1:
        print(-1)
    else:
        print(min_len)

if __name__ == "__main__":
    solve()
```
