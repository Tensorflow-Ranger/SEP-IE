```python
import sys
import math

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    K = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    max_val = max(A)
    freq = [0] * (max_val + 1)
    for x in A:
        freq[x] += 1

    # prefix sums of frequencies for multiples
    prefix = [0] * (max_val + 2)
    for d in range(1, max_val + 1):
        cnt = 0
        for m in range(d, max_val + 1, d):
            cnt += freq[m]
        prefix[d] = cnt

    # For each possible GCD d, how many numbers are multiples of d
    # prefix[d] already holds that.

    # For each i, we want the largest d such that:
    #   count_multiples_of_d >= K   (we can pick K numbers all multiples of d)
    #   and A[i] is a multiple of d.

    # Precompute for each d whether it's possible (count_multiples_of_d >= K)
    possible = [False] * (max_val + 1)
    for d in range(1, max_val + 1):
        if prefix[d] >= K:
            possible[d] = True

    # For each A[i], find max divisor d of A[i] with possible[d] == True
    # We can do this by checking all divisors of A[i] in O(sqrt(A_i))
    # But N up to 1.2e6 and A_i up to 1e6 -> total sqrt operations ~ 1.2e6 * 1e3 = 1.2e9 (too big)
    # Need faster: precompute for each number the largest possible divisor.

    # Instead, for each d where possible[d] is true, mark all multiples of d as "at least d"
    best = [0] * (max_val + 1)
    for d in range(1, max_val + 1):
        if possible[d]:
            for m in range(d, max_val + 1, d):
                best[m] = max(best[m], d)

    out = []
    for x in A:
        out.append(str(best[x]))
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```
