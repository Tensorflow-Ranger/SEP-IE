import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    H = int(next(it))
    W = int(next(it))
    rows = [int(next(it), 2) for _ in range(H)]

    # Precompute column masks
    col_masks = [1 << j for j in range(W)]

    # For each column subset, compute the best total
    best = H * W  # upper bound
    # We'll try all column flips (2^W possibilities)
    for col_mask in range(1 << W):
        total = 0
        # For each row, decide whether to flip it
        for row_val in rows:
            flipped = row_val ^ col_mask
            ones = bin(flipped).count("1")
            total += min(ones, W - ones)  # flip row if beneficial
        best = min(best, total)
    print(best)

if __name__ == "__main__":
    solve()
