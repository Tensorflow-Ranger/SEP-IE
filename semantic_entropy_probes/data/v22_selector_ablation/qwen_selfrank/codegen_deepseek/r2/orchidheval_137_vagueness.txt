def compare_one(a, b):
    """
    Create a function that takes different representing
    real numbers, and returns the larger variable in its given variable type.
    Return None if the values are equal.
    """
    # Helper to convert input to float for comparison
    def to_float(val):
        if isinstance(val, str):
            # Replace comma with dot for European decimal format
            return float(val.replace(',', '.'))
        return float(val)
    
    # Convert both to float for numeric comparison
    a_float = to_float(a)
    b_float = to_float(b)
    
    if a_float > b_float:
        return a
    elif b_float > a_float:
        return b
    else:
        return None
