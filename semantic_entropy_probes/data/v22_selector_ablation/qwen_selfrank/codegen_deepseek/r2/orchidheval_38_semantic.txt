def decode_cyclic(s: str):
    """
    takes as input string encoded with encode_cyclic function. Returns decoded string.
    """
    # Since encode_cyclic rotates each group of three one position left,
    # decoding is equivalent to rotating each group one position right,
    # which can be done by applying encode_cyclic twice.
    return encode_cyclic(encode_cyclic(s))
