```python
import sys
import math

def main():
    input = sys.stdin.readline
    N, K = map(int, input().split())
    A = list(map(int, input().split()))
    max_a = max(A)

    # Step 1: Frequency array
    freq = [0] * (max_a + 1)
    for x in A:
        freq[x] += 1

    # Step 2: Multiples count array
    multiples = [0] * (max_a + 1)
    for d in range(1, max_a + 1):
        cnt = 0
        for m in range(d, max_a + 1, d):
            cnt += freq[m]
        multiples[d] = cnt

    # Step 3: For each A_i, find largest divisor with multiples >= K
    ans = []
    for x in A:
        # We need K elements including x, so we need at least K-1 other elements
        # that are multiples of the divisor.
        best = 1
        # Iterate over divisors of x
        for d in range(1, int(math.isqrt(x)) + 1):
            if x % d == 0:
                # Check divisor d
                if multiples[d] >= K:
                    best = max(best, d)
                # Check paired divisor x // d
                d2 = x // d
                if d2 != d and multiples[d2] >= K:
                    best = max(best, d2)
        ans.append(best)

    sys.stdout.write("\n".join(map(str, ans)))

if __name__ == "__main__":
    main()
```
