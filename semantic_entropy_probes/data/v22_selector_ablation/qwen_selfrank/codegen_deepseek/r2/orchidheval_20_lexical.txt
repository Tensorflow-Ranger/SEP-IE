from typing import List, Tuple


def find_closest_elements(numbers: List[float]) -> Tuple[float, float]:
    """ From a supplied list of numbers (of length at least two) select and return two that are the nearest neighbors and return them in order (smaller number, larger number).
    """
    n = len(numbers)
    if n < 2:
        raise ValueError("List must contain at least two numbers")
    
    numbers_sorted = sorted(numbers)
    min_distance = float('inf')
    result = (numbers_sorted[0], numbers_sorted[1])
    
    for i in range(n - 1):
        dist = numbers_sorted[i + 1] - numbers_sorted[i]
        if dist < min_distance:
            min_distance = dist
            result = (numbers_sorted[i], numbers_sorted[i + 1])
    
    return result
