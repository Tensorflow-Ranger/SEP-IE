def digits(n):
    """Given a positive integer n, return the product of the key digits.
    Return 0 if all digits are even.
    """
    product = 1
    has_odd = False
    while n > 0:
        digit = n % 10
        if digit % 2 == 1:
            product *= digit
            has_odd = True
        n //= 10
    if not has_odd:
        return 0
    return product
