import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    A = [int(next(it)) for _ in range(N)]
    B = [int(next(it)) for _ in range(N)]
    C = [int(next(it)) for _ in range(N)]

    # We process bits from least significant (index N-1) to most significant (index 0)
    # because flipping a bit affects the cost of all more significant bits.
    total_cost = 0
    for i in range(N - 1, -1, -1):
        if A[i] != B[i]:
            total_cost += sum(C[j] for j in range(i, N) if A[j] == 1)
            # Flip A[i] (and implicitly all bits to the right remain unchanged for this operation)
            A[i] ^= 1
            # After flipping, we must also update the cost contributions of bits to the left (higher indices)
            # but since we process from right to left, those will be handled in later iterations.

    print(total_cost)

if __name__ == "__main__":
    solve()
