import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    t = int(next(it))
    out_lines = []
    for _ in range(t):
        n = int(next(it))
        A = [int(next(it)) for _ in range(2 * n)]
        pos = [[] for _ in range(n + 1)]
        for idx, val in enumerate(A):
            pos[val].append(idx)
        
        # Check adjacency and compute parity
        adj = [False] * (n + 1)
        parity = [0] * (n + 1)  # 0: even distance, 1: odd distance
        for x in range(1, n + 1):
            p1, p2 = pos[x]
            if abs(p1 - p2) == 1:
                adj[x] = True
            # parity of distance between occurrences
            parity[x] = (abs(p1 - p2) - 1) % 2
        
        # Count couples with even and odd parity among non-adjacent ones
        even_cnt = 0
        odd_cnt = 0
        for x in range(1, n + 1):
            if not adj[x]:
                if parity[x] == 0:
                    even_cnt += 1
                else:
                    odd_cnt += 1
        
        # Answer: choose two non-adjacent couples with same parity
        ans = (even_cnt * (even_cnt - 1) // 2) + (odd_cnt * (odd_cnt - 1) // 2)
        out_lines.append(str(ans))
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
