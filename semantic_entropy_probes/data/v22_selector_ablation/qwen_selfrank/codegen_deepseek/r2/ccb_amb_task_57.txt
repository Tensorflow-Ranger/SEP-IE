```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    arr = list(map(int, data[1:1+n]))
    arr.sort()
    
    max_val = arr[-1]
    present = [False] * (max_val + 1)
    for x in arr:
        present[x] = True
    
    count = 0
    # Iterate over possible B values
    for b in arr:
        # For each B, iterate over possible differences d = B - A = C - B
        # A = B - d, C = B + d, need A > 0 and C <= max_val
        # d can be at most min(B-1, max_val - B)
        max_d = min(b - 1, max_val - b)
        for d in range(1, max_d + 1):
            if present[b - d] and present[b + d]:
                count += 1
    print(count)

if __name__ == "__main__":
    main()
```
