def iscube(a):
    '''
    Write a function that takes an integer a and returns True 
    if this integer has an integer root.
    '''
    if a == 0:
        return True
    # Find the cube root of the absolute value
    root = round(abs(a) ** (1/3))
    # Check if cubing the candidate root gives back the original value
    return root ** 3 == abs(a) and (a > 0 or (a < 0 and root ** 3 == -a))
