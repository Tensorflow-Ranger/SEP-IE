```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    # total distinct count in whole array
    total_distinct = len(set(A))

    # prefix distinct counts
    prefix_distinct = [0] * (N - 1)
    seen = set()
    for i in range(N - 1):
        seen.add(A[i])
        prefix_distinct[i] = len(seen)

    # suffix distinct counts
    suffix_distinct = [0] * (N - 1)
    seen.clear()
    for i in range(N - 1, 0, -1):
        seen.add(A[i])
        suffix_distinct[i - 1] = len(seen)

    # compute maximum sum
    ans = 0
    for i in range(N - 1):
        ans = max(ans, prefix_distinct[i] + suffix_distinct[i])

    print(ans)

if __name__ == "__main__":
    solve()
```
