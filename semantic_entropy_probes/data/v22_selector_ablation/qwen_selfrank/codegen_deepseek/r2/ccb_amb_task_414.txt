import sys

MOD = 998244353

def modinv(x):
    return pow(x, MOD - 2, MOD)

def main():
    data = sys.stdin.read().strip().split()
    it = iter(data)
    H = int(next(it))
    W = int(next(it))
    n = H + W - 2
    k = H - 1

    # Precompute factorials and inverse factorials
    fact = [1] * (n + 1)
    inv_fact = [1] * (n + 1)
    for i in range(1, n + 1):
        fact[i] = fact[i - 1] * i % MOD
    inv_fact[n] = modinv(fact[n])
    for i in range(n, 0, -1):
        inv_fact[i - 1] = inv_fact[i] * i % MOD

    def comb(a, b):
        if b < 0 or b > a:
            return 0
        return fact[a] * inv_fact[b] % MOD * inv_fact[a - b] % MOD

    total_paths = comb(n, k)

    grid = []
    for _ in range(H):
        row = [int(next(it)) % MOD for _ in range(W)]
        grid.append(row)

    Q = int(next(it))
    sh = int(next(it)) - 1
    sw = int(next(it)) - 1

    # Precompute contributions of each cell
    # contribution[h][w] = comb(h + w, h) * comb((H-1 - h) + (W-1 - w), (H-1 - h))
    contrib = [[0] * W for _ in range(H)]
    for h in range(H):
        for w in range(W):
            ways_to = comb(h + w, h)
            ways_from = comb((H - 1 - h) + (W - 1 - w), H - 1 - h)
            contrib[h][w] = ways_to * ways_from % MOD

    # Initial sum
    current_sum = 0
    for h in range(H):
        for w in range(W):
            current_sum = (current_sum + grid[h][w] * contrib[h][w]) % MOD

    out_lines = []
    directions = {'L': (0, -1), 'R': (0, 1), 'U': (-1, 0), 'D': (1, 0)}

    for _ in range(Q):
        d = next(it)
        a = int(next(it)) % MOD
        dh, dw = directions[d]
        sh += dh
        sw += dw

        old_val = grid[sh][sw]
        if old_val != a:
            delta = (a - old_val) % MOD
            grid[sh][sw] = a
            current_sum = (current_sum + delta * contrib[sh][sw]) % MOD

        out_lines.append(str(current_sum))

    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    main()
