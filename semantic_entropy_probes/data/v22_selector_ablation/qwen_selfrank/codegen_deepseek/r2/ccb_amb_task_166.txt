```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    # We need the shortest subarray with a repeated value.
    # That means two equal numbers with minimal distance between them.
    # For each value, track its last seen index.
    last_seen = {}
    min_len = N + 1  # larger than any possible

    for i, val in enumerate(A):
        if val in last_seen:
            dist = i - last_seen[val] + 1
            if dist < min_len:
                min_len = dist
        last_seen[val] = i

    if min_len == N + 1:
        print(-1)
    else:
        print(min_len)

if __name__ == "__main__":
    solve()
```
