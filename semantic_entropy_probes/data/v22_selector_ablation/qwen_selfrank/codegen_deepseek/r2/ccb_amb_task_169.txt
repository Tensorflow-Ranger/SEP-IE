```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    arr = [int(next(it)) for _ in range(n)]

    # Dictionary to store the last seen index of each value
    last_seen = {}
    min_len = n + 1  # Initialize with a value larger than possible

    for i, val in enumerate(arr):
        if val in last_seen:
            # Subarray from last occurrence to current i (inclusive)
            length = i - last_seen[val] + 1
            if length < min_len:
                min_len = length
        last_seen[val] = i

    if min_len == n + 1:
        print(-1)
    else:
        print(min_len)

if __name__ == "__main__":
    solve()
```
