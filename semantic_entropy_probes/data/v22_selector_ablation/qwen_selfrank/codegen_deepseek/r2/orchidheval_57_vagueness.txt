def monotonic(l: list):
    """Return True is list elements exhibit a pattern of increase or decrease.
    """
    non_decreasing = all(l[i] <= l[i + 1] for i in range(len(l) - 1))
    non_increasing = all(l[i] >= l[i + 1] for i in range(len(l) - 1))
    return non_decreasing or non_increasing
