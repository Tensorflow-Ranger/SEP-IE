```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    arr = list(map(int, data[1:1+n]))
    arr.sort()

    max_val = arr[-1]
    count = [0] * (max_val + 1)
    for v in arr:
        count[v] = 1

    ans = 0
    for i in range(n):
        b = arr[i]
        step = 1
        while True:
            a = b - step
            c = b + step
            if a < 1 or c > max_val:
                break
            if count[a] and count[c]:
                ans += 1
            step += 1

    print(ans)

if __name__ == "__main__":
    main()
```
