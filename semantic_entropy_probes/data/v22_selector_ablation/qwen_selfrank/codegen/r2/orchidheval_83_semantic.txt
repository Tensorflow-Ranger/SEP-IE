def starts_one_ends(n):
    """
    Given a positive integer n, return the count of the numbers of n-digit
    positive integers that either start or end with the digit 1.
    """
    if n == 1:
        return 1
    # Numbers starting with 1: 1 followed by (n-1) digits (1-9)
    start_with_1 = 1 * 9 ** (n - 1)
    # Numbers ending with 1: (1-9) followed by (n-1) digits ending with 1
    end_with_1 = 9 * 10 ** (n - 2)
    # Subtract overlap (numbers that both start and end with 1)
    overlap = 1 * 10 ** (n - 2)
    return start_with_1 + end_with_1 - overlap
