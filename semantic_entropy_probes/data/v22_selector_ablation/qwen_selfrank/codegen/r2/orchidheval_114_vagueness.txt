def minSubArraySum(nums):
    """
    Given an array of integers nums, find the minimal sum of a sub-array
    of nums.
    """
    min_sum = float('inf')
    current_sum = 0
    for num in nums:
        current_sum = min(num, current_sum + num)
        min_sum = min(min_sum, current_sum)
    return min_sum
