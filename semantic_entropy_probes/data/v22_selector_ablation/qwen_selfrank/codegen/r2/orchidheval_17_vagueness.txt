from typing import List

def parse_music(music_string: str) -> List[int]:
    tokens = music_string.split()
    durations = []
    for token in tokens:
        if token == 'o':
            durations.append(4)
        elif token == 'o|':
            durations.append(2)
        elif token == '.|':
            durations.append(1)
    return durations
