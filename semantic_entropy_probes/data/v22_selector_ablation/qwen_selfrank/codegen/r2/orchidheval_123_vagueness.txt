def get_odd_collatz(n):
    collatz_sequence = []
    current = n
    while current != 1:
        if current % 2 == 1:
            collatz_sequence.append(current)
        if current % 2 == 0:
            current //= 2
        else:
            current = 3 * current + 1
    if current % 2 == 1:
        collatz_sequence.append(current)
    return sorted(collatz_sequence)
