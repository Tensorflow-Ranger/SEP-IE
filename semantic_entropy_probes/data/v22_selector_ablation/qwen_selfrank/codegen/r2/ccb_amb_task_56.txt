```python
import sys
import bisect

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    N = int(data[0])
    S = list(map(int, data[1:]))

    S.sort()
    count = 0
    for i in range(N):
        for j in range(i + 1, N):
            target = S[i] + S[j]
            k = bisect.bisect_left(S, target, j + 1)
            if k < N and S[k] == target:
                count += 1
    print(count)

if __name__ == "__main__":
    main()
```
