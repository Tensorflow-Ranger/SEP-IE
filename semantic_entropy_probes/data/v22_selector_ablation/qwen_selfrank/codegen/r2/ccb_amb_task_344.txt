```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    
    N = int(data[0])
    A = list(map(int, data[1:N+1]))
    B = list(map(int, data[N+1:2*N+1]))
    C = list(map(int, data[2*N+1:3*N+1]))
    
    cost = 0
    current_A = A[:]
    
    for i in range(N):
        if current_A[i] != B[i]:
            cost += sum(current_A) + sum(C)
            current_A[i] = 1 - current_A[i]
    
    print(cost)

if __name__ == "__main__":
    main()
```
