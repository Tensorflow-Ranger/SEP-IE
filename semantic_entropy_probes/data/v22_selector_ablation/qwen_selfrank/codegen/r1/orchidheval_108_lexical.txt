def count_nums(arr):
    import math
    count = 0
    for num in arr:
        digits_sum = 0
        num = abs(num)
        digits = []
        while num > 0:
            digits.append(num % 10)
            num = num // 10
        digits = digits[::-1]
        if digits:
            digits_sum += digits[0]
            for d in digits[1:]:
                digits_sum += d
        if digits_sum > 0:
            count += 1
    return count
