```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    N = int(data[0])
    A = list(map(int, data[1:]))

    distinct_left = [0] * N
    distinct_right = [0] * N

    seen = set()
    count = 0
    for i in range(N):
        if A[i] not in seen:
            seen.add(A[i])
            count += 1
        distinct_left[i] = count

    seen = set()
    count = 0
    for i in range(N-1, -1, -1):
        if A[i] not in seen:
            seen.add(A[i])
            count += 1
        distinct_right[i] = count

    max_sum = 0
    for i in range(N-1):
        current = distinct_left[i] + distinct_right[i+1]
        if current > max_sum:
            max_sum = current

    print(max_sum)

if __name__ == "__main__":
    main()
```
