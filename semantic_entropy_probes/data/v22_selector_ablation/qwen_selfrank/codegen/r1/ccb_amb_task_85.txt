def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    N = int(data[0])
    P = list(map(int, data[1:N+1]))
    Q = list(map(int, data[N+1:2*N+1]))
    
    # Create a mapping from bib number to person index
    bib_to_person = [0] * (N + 1)
    for idx in range(N):
        bib_to_person[Q[idx]] = idx
    
    # For each bib number i, find the person staring at i
    result = [0] * (N + 1)
    for i in range(1, N + 1):
        person = P[i - 1]
        result[i] = Q[bib_to_person[i]]
    
    print(' '.join(map(str, result[1:N+1])))

if __name__ == "__main__":
    main()
