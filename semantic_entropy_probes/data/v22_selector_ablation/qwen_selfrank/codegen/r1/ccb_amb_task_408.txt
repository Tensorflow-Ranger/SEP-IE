import sys
import bisect

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    N = int(data[0])
    M = int(data[1])
    ops = []
    for i in range(M):
        L = int(data[2 + 2*i])
        R = int(data[3 + 2*i])
        ops.append((L, R))
    
    # For each position, track the earliest and latest operation that can set it to 1
    earliest = [float('inf')] * (N + 2)
    latest = [-float('inf')] * (N + 2)
    
    for i in range(M):
        L, R = ops[i]
        # Operation 1: set [L, R] to 1
        earliest[L] = min(earliest[L], i)
        latest[R] = max(latest[R], i)
        # Operation 2: set [1, L-1] and [R+1, N] to 1
        earliest[1] = min(earliest[1], i)
        latest[L-1] = max(latest[L-1], i)
        earliest[R+1] = min(earliest[R+1], i)
        latest[N] = max(latest[N], i)
    
    # Now, check if all positions can be set to 1
    res = []
    for i in range(1, N+1):
        e = earliest[i]
        l = latest[i]
        if e == float('inf') or l == -float('inf'):
            print(-1)
            return
    
    # Reconstruct the operations
    used = [False] * M
    for i in range(1, N+1):
        e = earliest[i]
        l = latest[i]
        # Find an operation that can cover this position
        for j in range(e, l+1):
            if not used[j]:
                used[j] = True
                res.append(1 if ops[j][0] <= i <= ops[j][1] else 2)
                break
    
    print(len(res))
    print(' '.join(map(str, res)))

if __name__ == "__main__":
    main()
