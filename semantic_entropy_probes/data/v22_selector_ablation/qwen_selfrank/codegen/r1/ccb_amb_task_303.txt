```python
import sys
import sys
input = sys.stdin.read

def main():
    data = input().split()
    idx = 0
    T = int(data[idx])
    idx += 1
    results = []
    
    for _ in range(T):
        N = int(data[idx])
        idx += 1
        A = list(map(int, data[idx:idx + 2 * N]))
        idx += 2 * N
        
        pos = [0] * (N + 1)
        for i in range(2 * N):
            pos[A[i]] = i
        
        count = 0
        for a in range(1, N + 1):
            for b in range(a + 1, N + 1):
                a1, a2 = pos[a]
                b1, b2 = pos[b]
                if abs(a1 - a2) == 1 or abs(b1 - b2) == 1:
                    continue
                if (abs(a1 - b1) == 1 and abs(a2 - b2) == 1) or \
                   (abs(a1 - b2) == 1 and abs(a2 - b1) == 1):
                    count += 1
        results.append(str(count))
    
    print('\n'.join(results))
```
