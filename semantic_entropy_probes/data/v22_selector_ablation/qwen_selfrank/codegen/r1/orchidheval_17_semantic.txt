from typing import List

def parse_music(music_string: str) -> List[int]:
    notes = music_string.split()
    result = []
    for note in notes:
        if note == 'o':
            result.append(1)
        elif note == 'o|':
            result.append(2)
        elif note == 'x':
            result.append(3)
    return result
