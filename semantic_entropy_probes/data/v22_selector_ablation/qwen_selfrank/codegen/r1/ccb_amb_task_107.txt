```python
import sys
import math
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    N = int(data[0])
    K = int(data[1])
    A = list(map(int, data[2:2+N]))
    
    max_val = max(A)
    freq = defaultdict(int)
    for num in A:
        freq[num] += 1
    
    # Precompute GCDs for all possible subsets
    # We use inclusion-exclusion and sieve-like approach
    # For each d from max_val down to 1
    # Count how many numbers are divisible by d
    # Then compute how many subsets of size K have GCD exactly d
    
    count = [0] * (max_val + 1)
    for d in range(1, max_val + 1):
        cnt = 0
        for multiple in range(d, max_val + 1, d):
            cnt += freq[multiple]
        count[d] = cnt
    
    # Now compute the number of subsets with GCD exactly d
    # Using inclusion-exclusion
    exact = [0] * (max_val + 1)
    for d in range(max_val, 0, -1):
        exact[d] = count[d]
        for multiple in range(2 * d, max_val + 1, d):
            exact[d] -= exact[multiple]
    
    # For each element A[i], find the maximum d such that there are at least K elements divisible by d
    # and the exact count of subsets with GCD exactly d is non-zero
    result = []
    for num in A:
        max_gcd = 0
        for d in range(num, 0, -1):
            if exact[d] >= 1:
                max_gcd = d
                break
        result.append(str(max_gcd))
    
    print(' '.join(result))

if __name__ == "__main__":
    main()
```
