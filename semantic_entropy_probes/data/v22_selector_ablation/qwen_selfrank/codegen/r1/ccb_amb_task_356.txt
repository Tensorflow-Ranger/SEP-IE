```python
import sys
import heapq
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    N = int(data[idx])
    idx += 1
    
    W = list(map(int, data[idx:idx+N]))
    idx += N
    
    intervals = []
    for _ in range(N):
        L = int(data[idx])
        R = int(data[idx+1])
        intervals.append((L, R))
        idx += 2
    
    Q = int(data[idx])
    idx += 1
    
    queries = []
    for _ in range(Q):
        s = int(data[idx])
        t = int(data[idx+1])
        queries.append((s, t))
        idx += 2
    
    # Build adjacency list for intervals
    adj = [[] for _ in range(N)]
    for i in range(N):
        L, R = intervals[i]
        for j in range(N):
            if i == j:
                continue
            Lj, Rj = intervals[j]
            if not (R < Lj or Rj < L):
                adj[i].append(j)
    
    # Dijkstra's algorithm for each query
    import sys
    import heapq
    
    def dijkstra(start, end):
        dist = [float('inf')] * N
        dist[start] = W[start - 1]
        heap = [(W[start - 1], start)]
        visited = [False] * N
        
        while heap:
            cost, u = heapq.heappop(heap)
            if visited[u]:
                continue
            visited[u] = True
            if u == end:
                return cost
            for v in adj[u]:
                if not visited[v]:
                    new_cost = cost + W[v - 1]
                    if new_cost < dist[v]:
                        dist[v] = new_cost
                        heapq.heappush(heap, (new_cost, v))
        return -1
    
    for s, t in queries:
        print(dijkstra(s - 1, t - 1))

if __name__ == "__main__":
    main()
```
