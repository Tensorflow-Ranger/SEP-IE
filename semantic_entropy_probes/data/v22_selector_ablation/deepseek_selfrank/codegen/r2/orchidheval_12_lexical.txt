from typing import List, Optional

def longest(strings: List[str]) -> Optional[str]:
    """ Out of list of strings, return the longest one. Return the leading one in case of multiple
    strings of the same length. Return None in case the input list is empty.
    """
    if not strings:
        return None
    max_len = -1
    result = None
    for s in strings:
        if len(s) > max_len:
            max_len = len(s)
            result = s
    return result
