def starts_one_ends(n):
    """
    Given a positive integer n, return the count of the numbers of n-digit
    positive integers that either start or end with the digit 1.
    """
    if n == 1:
        return 1
    start_with_one = 1 * 10**(n-1)
    end_with_one = 9 * 10**(n-2)
    both_start_and_end_with_one = 1 * 10**(n-2)
    return start_with_one + end_with_one - both_start_and_end_with_one
