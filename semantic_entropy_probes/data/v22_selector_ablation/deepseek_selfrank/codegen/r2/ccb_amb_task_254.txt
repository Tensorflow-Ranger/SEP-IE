```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    N = int(data[0])
    A = list(map(int, data[1:]))
    
    prefix = [0] * (N + 1)
    suffix = [0] * (N + 1)
    
    distinct = set()
    for i in range(N):
        distinct.add(A[i])
        prefix[i + 1] = len(distinct)
    
    distinct = set()
    for i in range(N - 1, -1, -1):
        distinct.add(A[i])
        suffix[i] = len(distinct)
    
    max_sum = 0
    for i in range(1, N):
        current = prefix[i] + suffix[i]
        if current > max_sum:
            max_sum = current
    print(max_sum)

if __name__ == "__main__":
    main()
```
