def prime_length(string):
    """Write a function that takes a string and returns True if the number of individual characters is a prime or False otherwise
    """
    length = len(string)
    if length < 2:
        return False
    for i in range(2, int(length**0.5) + 1):
        if length % i == 0:
            return False
    return True
