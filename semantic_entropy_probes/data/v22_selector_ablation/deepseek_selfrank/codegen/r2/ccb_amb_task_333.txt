```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    N = int(data[0])
    P = list(map(int, data[1:N+1]))
    
    pos = [0] * (N + 1)
    for i in range(N):
        pos[P[i]] = i
    
    res = 0
    for i in range(1, N + 1):
        if pos[i] != i - 1:
            res += pos[i] + 1
            pos[i + 1] += 1
    
    print(res)

if __name__ == "__main__":
    main()
```
