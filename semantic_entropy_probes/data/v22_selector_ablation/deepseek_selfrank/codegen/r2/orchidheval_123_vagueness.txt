def get_odd_collatz(n):
    import math
    if n <= 0:
        return []
    collatz = []
    current = n
    while current != 1:
        if current % 2 == 1:
            collatz.append(current)
        if current % 2 == 0:
            current //= 2
        else:
            current = 3 * current + 1
    if 1 % 2 == 1:
        collatz.append(1)
    return sorted(collatz)
