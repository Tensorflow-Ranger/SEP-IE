```python
import sys
import bisect

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    N = int(data[0])
    M = int(data[1])
    A = list(map(int, data[2:2+N]))
    
    inv_count = [0] * M
    for k in range(M):
        B = [(a + k) % M for a in A]
        inv = 0
        for i in range(N):
            for j in range(i + 1, N):
                if B[i] > B[j]:
                    inv += 1
        inv_count[k] = inv
    
    for count in inv_count:
        print(count)

if __name__ == "__main__":
    main()
```
