import sys
import bisect

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    N = int(data[0])
    A = list(map(int, data[1:]))

    prefix = [0] * (N + 1)
    suffix = [0] * (N + 1)
    distinct = set()

    # Compute prefix distinct counts
    for i in range(N):
        distinct.add(A[i])
        prefix[i + 1] = len(distinct)

    distinct = set()
    # Compute suffix distinct counts
    for i in range(N - 1, -1, -1):
        distinct.add(A[i])
        suffix[i] = len(distinct)

    max_sum = 0
    # For each possible split i (split after i)
    for i in range(1, N - 1):
        # For each possible split j (split after j)
        for j in range(i + 1, N):
            sum_counts = prefix[i] + suffix[j] + (N - j)
            if sum_counts > max_sum:
                max_sum = sum_counts

    print(max_sum)

if __name__ == "__main__":
    main()
