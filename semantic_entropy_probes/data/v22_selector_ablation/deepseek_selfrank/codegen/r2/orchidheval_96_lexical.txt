def count_up_to(n):
    """Implement a function that takes an non-negative integer and returns an array of the top n
    integers that are prime numbers and less than n.
    """
    if n <= 0:
        return []
    primes = []
    candidate = 2
    while len(primes) < n:
        is_prime = True
        for p in primes:
            if p * p > candidate:
                break
            if candidate % p == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(candidate)
        candidate += 1
    return primes
