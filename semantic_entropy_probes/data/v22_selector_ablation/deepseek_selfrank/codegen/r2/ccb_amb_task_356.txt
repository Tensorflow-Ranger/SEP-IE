```python
import sys
import heapq
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    N = int(data[idx])
    idx += 1
    
    W = list(map(int, data[idx:idx+N]))
    idx += N
    
    intervals = []
    for _ in range(N):
        L = int(data[idx])
        R = int(data[idx+1])
        intervals.append((L, R))
        idx += 2
    
    Q = int(data[idx])
    idx += 1
    
    queries = []
    for _ in range(Q):
        s = int(data[idx])
        t = int(data[idx+1])
        queries.append((s, t))
        idx += 2
    
    # Build adjacency list
    adj = [[] for _ in range(N)]
    for i in range(N):
        L_i, R_i = intervals[i]
        for j in range(i+1, N):
            L_j, R_j = intervals[j]
            if not (R_i < L_j or R_j < L_i):
                adj[i].append(j)
                adj[j].append(i)
    
    # Dijkstra's algorithm for each query
    import math
    INF = math.inf
    
    for s, t in queries:
        dist = [INF] * N
        dist[s-1] = W[s-1]
        heap = [(W[s-1], s-1)]
        
        while heap:
            curr_w, u = heapq.heappop(heap)
            if curr_w > dist[u]:
                continue
            for v in adj[u]:
                if dist[v] > dist[u] + W[v]:
                    dist[v] = dist[u] + W[v]
                    heapq.heappush(heap, (dist[v], v))
        
        if dist[t-1] == INF:
            print(-1)
        else:
            print(dist[t-1])

if __name__ == "__main__":
    main()
```
