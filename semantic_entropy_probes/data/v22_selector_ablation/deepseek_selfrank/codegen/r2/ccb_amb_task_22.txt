import sys
import bisect

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    N = int(data[idx])
    idx += 1
    W = int(data[idx])
    idx += 1
    
    # For each column, store sorted list of Y coordinates of blocks
    col_y = [[] for _ in range(W + 1)]
    for i in range(N):
        X = int(data[idx])
        idx += 1
        Y = int(data[idx])
        idx += 1
        col_y[X].append(Y)
    
    # Sort each column's Y values
    for i in range(1, W + 1):
        col_y[i].sort()
    
    Q = int(data[idx])
    idx += 1
    
    queries = []
    for _ in range(Q):
        T = int(data[idx])
        idx += 1
        A = int(data[idx]) - 1  # Convert to 0-based index
        idx += 1
        queries.append((T, A))
    
    # For each query, determine if block A exists at time T
    results = []
    
    for T, A in queries:
        X = A // W + 1  # Compute column index
        Y = A % W + 1   # Compute original Y coordinate (but we need to track movement)
        
        # Since blocks are moved down over time, we can model the movement
        # For a block starting at (X, Y), after t steps, it would have moved down t times
        # However, blocks are removed when the bottom row is filled, so we need to simulate the process
        
        # Instead of simulating, we can find how many blocks are below the original Y
        # and determine if the block would have been removed
        # We can use binary search to find the number of blocks in the same column that are below Y
        
        # Since the blocks are sorted, we can find the number of blocks in column X that are <= Y
        # This gives the number of blocks that would have been removed before this block
        # If the number of such blocks is equal to the number of steps, then the block is removed
        
        # But since the blocks are processed in order, and each step removes the bottom row,
        # the number of blocks that can be removed from column X is the number of blocks in that column
        # So for each block, the number of steps it can survive is the number of blocks in its column that are above it
        
        # So for block A, which starts at (X, Y), the number of blocks in column X that are above it is
        # len(col_y[X]) - bisect.bisect_right(col_y[X], Y)
        
        # If T is greater than this number, the block is removed
        # Otherwise, it's still present
        
        col = col_y[X]
        pos = bisect.bisect_right(col, Y)
        num_blocks_above = len(col) - pos
        
        if T > num_blocks_above:
            results.append("No")
        else:
            results.append("Yes")
    
    print("\n".join(results))

if __name__ == "__main__":
    main()
