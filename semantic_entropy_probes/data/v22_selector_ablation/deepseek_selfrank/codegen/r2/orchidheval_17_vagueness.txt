from typing import List

def parse_music(music_string: str) -> List[int]:
    durations = music_string.split()
    result = []
    for note in durations:
        if note == 'o':
            result.append(4)
        elif note == 'o|':
            result.append(2)
        elif note == '.|':
            result.append(1)
    return result
