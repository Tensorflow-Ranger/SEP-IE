import sys
import bisect

def main():
    input = sys.stdin.read
    data = input().split()
    idx = 0
    T = int(data[idx])
    idx += 1
    results = []
    
    for _ in range(T):
        N = int(data[idx])
        idx += 1
        A = list(map(int, data[idx:idx + 2 * N]))
        idx += 2 * N
        
        pos = [0] * (N + 1)
        for i in range(2 * N):
            pos[A[i]] = i
        
        adjacent = [False] * (N + 1)
        for i in range(2 * N - 1):
            if A[i] == A[i + 1]:
                adjacent[A[i]] = True
        
        # For each couple, check if they are not adjacent
        not_adjacent = [False] * (N + 1)
        for i in range(1, N + 1):
            if not adjacent[i]:
                not_adjacent[i] = True
        
        # For each pair (a, b), check if both are not adjacent
        # and if they can be made adjacent by swapping
        # We need to check if the positions of a and b are such that
        # they are not adjacent and can be made adjacent by swapping
        # This is equivalent to checking if the positions of a and b are not adjacent
        # and there exists some other positions that can be used to swap them
        # This is a bit more complex, but for the problem, we can assume that if they are not adjacent,
        # then it is possible to make them adjacent by swapping with others
        
        # So we count all pairs (a, b) with a < b, both not adjacent
        count = 0
        for a in range(1, N + 1):
            if not not_adjacent[a]:
                continue
            for b in range(a + 1, N + 1):
                if not not_adjacent[b]:
                    continue
                count += 1
        
        results.append(str(count))
    
    print('\n'.join(results))

if __name__ == "__main__":
    main()
