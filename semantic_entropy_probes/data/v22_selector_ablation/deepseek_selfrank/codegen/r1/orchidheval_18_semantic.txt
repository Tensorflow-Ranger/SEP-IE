def how_many_times(string: str, substring: str) -> int:
    """ Determine the number of instances a given substring can be found in the original string. 
    """
    count = 0
    start = 0
    while True:
        start = string.find(substring, start)
        if start == -1:
            break
        count += 1
        start += 1
    return count
