```python
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    N = int(data[0])
    P = list(map(int, data[1:]))

    pos = [0] * (N + 1)
    for i in range(N):
        pos[P[i]] = i

    cost = 0
    for i in range(1, N + 1):
        if pos[i] == i:
            continue
        j = pos[i]
        cost += j
        pos[P[j]] = i
        pos[i] = j
        if j > i:
            cost += j - i
            pos[P[j]] = i
            pos[i] = j
        else:
            cost += j
            pos[P[j]] = i
            pos[i] = j

    print(cost)

if __name__ == "__main__":
    main()
```
