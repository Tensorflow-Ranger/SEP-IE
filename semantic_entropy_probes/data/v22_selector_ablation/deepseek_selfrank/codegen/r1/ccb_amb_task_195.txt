```python
import sys
from itertools import product

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    H = int(data[0])
    W = int(data[1])
    grid = data[2:2+H]
    
    # Convert each row to a list of integers
    grid = [list(map(int, list(row))) for row in grid]
    
    # Calculate initial sum
    total = sum(sum(row) for row in grid)
    
    # Try all possible combinations of flipping columns (since W <= 18, 2^18 is manageable)
    for mask in range(1 << W):
        # For each column, decide whether to flip it
        new_total = total
        for j in range(W):
            if (mask >> j) & 1:
                # Flip column j
                count = sum(grid[i][j] for i in range(H))
                new_total += H - 2 * count
        # Keep the minimum total
        if new_total < total:
            total = new_total
    
    print(total)

if __name__ == "__main__":
    main()
```
