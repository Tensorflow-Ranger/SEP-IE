```python
import sys
import bisect

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    N = int(data[idx])
    idx += 1
    W = int(data[idx])
    idx += 1
    
    blocks = [[] for _ in range(W + 1)]
    for i in range(N):
        X = int(data[idx])
        idx += 1
        Y = int(data[idx])
        idx += 1
        blocks[X].append(Y)
    
    for x in range(1, W + 1):
        blocks[x].sort()
    
    Q = int(data[idx])
    idx += 1
    
    queries = []
    for _ in range(Q):
        T = int(data[idx])
        idx += 1
        A = int(data[idx]) - 1
        idx += 1
        queries.append((T, A))
    
    result = []
    
    for T, A in queries:
        x = A // 1000000
        y = A % 1000000
        if x >= W or x < 0:
            result.append("No")
            continue
        ys = blocks[x]
        if not ys:
            result.append("No")
            continue
        pos = bisect.bisect_left(ys, y)
        if pos >= len(ys) or ys[pos] != y:
            result.append("No")
            continue
        
        # Calculate how many times the block has fallen
        cnt = 0
        for i in range(pos, len(ys)):
            if ys[i] - ys[pos] > cnt:
                cnt = ys[i] - ys[pos]
        
        if cnt < T:
            result.append("No")
        else:
            result.append("Yes")
    
    print("\n".join(result))

if __name__ == "__main__":
    main()
```
