def count_nums(arr):
    import math
    count = 0
    for num in arr:
        total = 0
        if num < 0:
            num = -num
            total -= 1  # account for the negative sign
        while num > 0:
            total += num % 10
            num //= 10
        if total > 0:
            count += 1
    return count
