import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    N = int(data[0])
    A = list(map(int, data[1:]))

    prefix = [0] * (N + 1)
    suffix = [0] * (N + 1)
    count = {}

    # Compute prefix distinct counts
    for i in range(N):
        if A[i] not in count:
            count[A[i]] = 0
        count[A[i]] += 1
        prefix[i + 1] = prefix[i] + (1 if count[A[i]] == 1 else 0)

    count = {}
    # Compute suffix distinct counts
    for i in range(N - 1, -1, -1):
        if A[i] not in count:
            count[A[i]] = 0
        count[A[i]] += 1
        suffix[i] = suffix[i + 1] + (1 if count[A[i]] == 1 else 0)

    max_sum = 0
    for i in range(1, N - 1):
        for j in range(i + 1, N):
            current = prefix[i] + suffix[j + 1]
            if current > max_sum:
                max_sum = current
    print(max_sum)

if __name__ == "__main__":
    main()
