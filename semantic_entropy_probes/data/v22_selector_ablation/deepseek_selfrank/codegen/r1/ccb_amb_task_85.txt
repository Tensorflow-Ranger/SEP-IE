```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    
    N = int(data[0])
    P = list(map(int, data[1:N+1]))
    Q = list(map(int, data[N+1:2*N+1]))
    
    result = [0] * N
    for i in range(N):
        person = Q[i] - 1  # Convert to 0-based index
        result[person] = P[i]
    
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
```
