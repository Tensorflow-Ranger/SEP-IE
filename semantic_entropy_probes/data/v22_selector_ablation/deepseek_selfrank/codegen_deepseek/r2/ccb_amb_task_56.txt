```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    arr = list(map(int, data[1:1 + n]))
    arr.sort()

    max_val = arr[-1]
    present = [False] * (max_val + 1)
    for x in arr:
        present[x] = True

    count = 0
    # Iterate over possible A and C, compute B
    for i in range(n):
        A = arr[i]
        for j in range(i + 1, n):
            C = arr[j]
            B = A + C
            if B % 2 == 0:
                B //= 2
                if B > A and B < C and B <= max_val and present[B]:
                    count += 1
    print(count)

if __name__ == "__main__":
    main()
```
