def triangle_area(a, b, c):
    '''
    Given the lengths of the three sides of a triangle. Return the area of
    the triangle rounded to 2 decimal points if the three sides form a triangle. 
    Otherwise return -1
    Three sides make a triangle when the sum of any two sides is greater 
    than the third side.
    '''
    # Check for non-positive sides or invalid triangle inequality
    if a <= 0 or b <= 0 or c <= 0:
        return -1
    if not (a + b > c and a + c > b and b + c > a):
        return -1
    
    # Heron's formula
    s = (a + b + c) / 2
    area = (s * (s - a) * (s - b) * (s - c)) ** 0.5
    return round(area, 2)
