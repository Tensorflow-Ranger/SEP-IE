```python
import sys
import math

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    K = int(next(it))
    A = [int(next(it)) for _ in range(N)]

    max_a = max(A)
    freq = [0] * (max_a + 1)
    for v in A:
        freq[v] += 1

    # prefix sums of counts for multiples
    prefix = [0] * (max_a + 1)
    for d in range(1, max_a + 1):
        cnt = 0
        for m in range(d, max_a + 1, d):
            cnt += freq[m]
        prefix[d] = cnt

    # For each i we need the largest d such that prefix[d] >= K and d divides A[i]
    # We can precompute the answer for each possible value of A[i]
    ans_for_val = [0] * (max_a + 1)
    for d in range(max_a, 0, -1):
        if prefix[d] >= K:
            # All multiples of d can have at least this gcd candidate
            for m in range(d, max_a + 1, d):
                if ans_for_val[m] == 0:
                    ans_for_val[m] = d

    out_lines = []
    for v in A:
        out_lines.append(str(ans_for_val[v]))
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    main()
```
