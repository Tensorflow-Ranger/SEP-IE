def compare_one(a, b):
    """
    Create a function that takes different representing
    real numbers, and returns the larger variable in its given variable type.
    Return None if the values are equal.
    """
    def to_float(val):
        if isinstance(val, str):
            # Replace comma with dot for decimal parsing
            val = val.replace(',', '.')
        return float(val)
    
    a_float = to_float(a)
    b_float = to_float(b)
    
    if a_float == b_float:
        return None
    elif a_float > b_float:
        return a
    else:
        return b
