```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    arr = list(map(int, data[1:1+n]))
    arr.sort()
    
    max_val = arr[-1]
    present = [False] * (max_val + 1)
    for x in arr:
        present[x] = True
    
    count = 0
    # Iterate over possible B values
    for i in range(1, n-1):
        B = arr[i]
        # Iterate over possible A values left of B
        left = i - 1
        while left >= 0:
            A = arr[left]
            diff = B - A
            C = B + diff
            if C > max_val:
                break
            if present[C]:
                count += 1
            left -= 1
    
    print(count)

if __name__ == "__main__":
    main()
```
