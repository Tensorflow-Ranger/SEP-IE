from typing import List

def parse_music(music_string: str) -> List[int]:
    note_durations = {
        'o': 4,
        'o|': 2,
        '.|': 1
    }
    result = []
    i = 0
    while i < len(music_string):
        if music_string[i] == 'o':
            if i + 1 < len(music_string) and music_string[i + 1] == '|':
                result.append(note_durations['o|'])
                i += 2
            else:
                result.append(note_durations['o'])
                i += 1
        elif music_string[i] == '.':
            result.append(note_durations['.|'])
            i += 2
        else:
            i += 1
    return result
