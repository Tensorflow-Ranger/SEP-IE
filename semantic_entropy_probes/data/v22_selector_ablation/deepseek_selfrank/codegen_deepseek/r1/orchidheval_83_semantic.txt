def starts_one_ends(n):
    """
    Given a positive integer n, return the count of the numbers of n-digit
    positive integers that either start or end with the digit 1.
    """
    if n <= 0:
        return 0
    # Total n-digit numbers: from 10^(n-1) to 10^n - 1
    # Count numbers starting with 1: first digit fixed as 1, remaining n-1 digits any of 0-9
    start_with_one = 10 ** (n - 1)
    # Count numbers ending with 1: last digit fixed as 1, first digit cannot be 0 (1-9), middle n-2 digits any of 0-9
    end_with_one = 9 * 10 ** (n - 2) if n >= 2 else 1
    # Count numbers both starting and ending with 1: first digit 1, last digit 1, middle n-2 digits any of 0-9
    both = 10 ** (n - 2) if n >= 2 else 1
    # Inclusion-exclusion
    return start_with_one + end_with_one - both
