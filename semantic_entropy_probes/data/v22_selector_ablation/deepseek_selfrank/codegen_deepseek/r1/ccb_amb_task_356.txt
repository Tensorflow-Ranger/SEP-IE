import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    it = iter(data)

    N = int(next(it))
    W = [0] + [int(next(it)) for _ in range(N)]

    intervals = [(0, 0)]  # dummy for 1‑based indexing
    for _ in range(N):
        L = int(next(it))
        R = int(next(it))
        intervals.append((L, R))

    # Build interval graph connectivity via sweep line
    events = []
    for i in range(1, N + 1):
        L, R = intervals[i]
        events.append((L, 0, i))   # open
        events.append((R, 1, i))   # close

    events.sort(key=lambda x: (x[0], x[1]))

    parent = list(range(N + 1))
    size = [1] * (N + 1)
    comp_min_weight = [10**18] * (N + 1)

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra == rb:
            return
        if size[ra] < size[rb]:
            ra, rb = rb, ra
        parent[rb] = ra
        size[ra] += size[rb]
        comp_min_weight[ra] = min(comp_min_weight[ra], comp_min_weight[rb])

    active = []
    for _, typ, idx in events:
        if typ == 0:  # open
            for act in active:
                union(idx, act)
            active.append(idx)
        else:  # close
            active.remove(idx)

    # After union, compute component minimum vertex weight
    for i in range(1, N + 1):
        r = find(i)
        if W[i] < comp_min_weight[r]:
            comp_min_weight[r] = W[i]

    Q = int(next(it))
    out_lines = []
    for _ in range(Q):
        s = int(next(it))
        t = int(next(it))
        if find(s) != find(t):
            out_lines.append("-1")
        else:
            root = find(s)
            out_lines.append(str(comp_min_weight[root]))

    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
