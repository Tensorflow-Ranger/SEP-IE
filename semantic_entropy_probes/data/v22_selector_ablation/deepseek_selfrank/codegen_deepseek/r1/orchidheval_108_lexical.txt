def count_nums(arr):
    """
    Write a function count_nums which takes an array of integers and returns
    the number of values which has a sum of digits > 0.
    If a number is negative, then its first signed digit will be negative:
    e.g. -123 has signed digits -1, 2, and 3.
    """
    def digit_sum(num):
        if num == 0:
            return 0
        sign = -1 if num < 0 else 1
        num = abs(num)
        total = 0
        first = True
        while num > 0:
            digit = num % 10
            if first:
                total += sign * digit
                first = False
            else:
                total += digit
            num //= 10
        return total

    count = 0
    for n in arr:
        if digit_sum(n) > 0:
            count += 1
    return count
