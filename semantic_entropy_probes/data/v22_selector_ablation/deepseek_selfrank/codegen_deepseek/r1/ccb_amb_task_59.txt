```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    arr = list(map(int, data[1:1 + n]))
    max_val = max(arr) if arr else 0
    present = [False] * (max_val + 1)
    for x in arr:
        present[x] = True

    count = 0
    # Iterate over possible B values
    for b in arr:
        # For each B, try possible differences d such that A = B - d, C = B + d
        # and A, C are within bounds and present.
        # We only need to check d up to min(b - 1, max_val - b)
        max_d = min(b - 1, max_val - b)
        for d in range(1, max_d + 1):
            a = b - d
            c = b + d
            if present[a] and present[c]:
                count += 1
    print(count)

if __name__ == "__main__":
    solve()
```
