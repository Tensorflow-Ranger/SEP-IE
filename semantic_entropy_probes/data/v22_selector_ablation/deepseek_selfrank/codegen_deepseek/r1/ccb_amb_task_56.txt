```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    arr = list(map(int, data[1:1 + n]))
    arr.sort()
    
    max_val = arr[-1]
    present = [0] * (max_val + 1)
    for x in arr:
        present[x] = 1
    
    count = 0
    for i in range(n - 2):
        A = arr[i]
        for j in range(i + 1, n - 1):
            B = arr[j]
            diff = B - A
            C = B + diff
            if C > max_val:
                break
            if present[C]:
                count += 1
    print(count)

if __name__ == "__main__":
    main()
```
