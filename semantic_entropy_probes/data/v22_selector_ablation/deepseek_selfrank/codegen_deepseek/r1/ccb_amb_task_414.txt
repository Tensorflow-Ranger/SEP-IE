import sys
sys.setrecursionlimit(1 << 25)

MOD = 998244353

def modinv(x):
    return pow(x, MOD - 2, MOD)

def main():
    data = sys.stdin.read().split()
    it = iter(data)
    H = int(next(it))
    W = int(next(it))
    n = H * W
    A = [0] * n
    for i in range(n):
        A[i] = int(next(it)) % MOD

    Q = int(next(it))
    sh = int(next(it)) - 1
    sw = int(next(it)) - 1

    # Precompute factorials and inverse factorials
    max_n = H + W + 5
    fact = [1] * (max_n)
    inv_fact = [1] * (max_n)
    for i in range(1, max_n):
        fact[i] = fact[i-1] * i % MOD
    inv_fact[max_n-1] = modinv(fact[max_n-1])
    for i in range(max_n-2, -1, -1):
        inv_fact[i] = inv_fact[i+1] * (i+1) % MOD

    def comb(nn, kk):
        if kk < 0 or kk > nn:
            return 0
        return fact[nn] * inv_fact[kk] % MOD * inv_fact[nn - kk] % MOD

    # Precompute binomial coefficients for all (dx, dy) with dx+dy = total steps
    # We'll store for each (dx, dy) the number of ways to go from (0,0) to (dx,dy)
    # Actually we need for each cell (h,w) the number of paths from (0,0) to (h,w) times
    # number of paths from (h,w) to (H-1,W-1). That's comb(h+w, h) * comb((H-1-h)+(W-1-w), H-1-h)
    # We'll precompute for all h,w.
    ways_to = [[0]*W for _ in range(H)]
    ways_from = [[0]*W for _ in range(H)]
    for h in range(H):
        for w in range(W):
            ways_to[h][w] = comb(h + w, h)
            ways_from[h][w] = comb((H-1 - h) + (W-1 - w), H-1 - h)

    # Total sum = sum over all cells (h,w) of A[h][w] * ways_to[h][w] * ways_from[h][w]
    total = 0
    for h in range(H):
        for w in range(W):
            total = (total + A[h*W + w] * ways_to[h][w] % MOD * ways_from[h][w]) % MOD

    out_lines = []
    pos = sh * W + sw
    for _ in range(Q):
        d = next(it)
        a = int(next(it)) % MOD
        # Move
        if d == 'L':
            sw -= 1
        elif d == 'R':
            sw += 1
        elif d == 'U':
            sh -= 1
        else:  # 'D'
            sh += 1
        new_pos = sh * W + sw
        # Update total
        old_val = A[pos]
        new_val = a
        coeff = ways_to[sh][sw] * ways_from[sh][sw] % MOD
        total = (total - old_val * coeff + new_val * coeff) % MOD
        A[new_pos] = new_val
        pos = new_pos
        out_lines.append(str(total))

    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    main()
