```python
def digits(n):
    """Given a positive integer n, return the product of the key digits.
    Return 0 if all digits are even.
    """
    product = 1
    has_odd = False
    
    for ch in str(n):
        digit = int(ch)
        if digit % 2 == 1:  # odd digit
            product *= digit
            has_odd = True
    
    if not has_odd:
        return 0
    return product
```
